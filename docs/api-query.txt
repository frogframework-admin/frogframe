RINGKASAN
---------
GET /api/query/{slug}?Param=nilai — query SQL read-only (mesin yang sama dengan
dashboard & laporan: tenant terisolasi, READ ONLY, statement_timeout, parameter
native) diterbitkan sebagai API. Tidak ada kode yang dieksekusi dari metadata.

Dua cara, satu jalur:
  1. Menu "API Query" → wizard: Query → Parameter & pratinjau → Terbitkan.
  2. Centang "Terbitkan juga sebagai API" di langkah simpan widget dashboard atau
     template laporan → query & parameter yang sama diterbitkan, ikut diperbarui
     setiap kali sumbernya disimpan; centang dilepas → API dihapus.

Opsi saat terbit: keluaran (daftar baris JSON / satu objek / CSV), batas baris
(maks. 10.000, truncated: true), cache 0–3600 detik, role pemanggil, aktif/nonaktif.
Parameter "dari akun" (departemen/ID pengguna) tidak bisa ditimpa pemanggil.
Kunci API: cakupan query:{slug} (dipilih di halaman Kunci API).

SUDAH DIUJI
  - 3 test layanan (PostgreSQL + read-only): terbit & dipanggil dengan parameter,
    argumen asing/tipe salah → 422, role → 403, nonaktif/tidak ada → 404, slug
    tidak valid/ganda, query non-SELECT ditolak saat terbit; parameter dari akun
    tidak bisa ditimpa (manager IT → data IT, manager Marketing → data Marketing);
    terbit dari widget bertaut (simpan ulang → API yang sama v2, slug milik sumber
    lain ditolak); cache menghemat eksekusi (dihitung), versi baru membatalkan cache.
  - Total: 52 lulus (in-memory) / 74 lulus (PostgreSQL).
  - Uji browser + HTTP 10 langkah (aplikasi berjalan, PostgreSQL): wizard API
    Query; opsi di wizard widget menerbitkan API bersumber "Widget dashboard";
    kunci dewi dengan cakupan satu query → 200 hanya data Marketing; parameter
    asing → 422; API lain → 403; daftar API untuk kunci hanya yang dalam cakupan;
    keluaran CSV text/csv; API dinonaktifkan → 404.

using System.Collections.Concurrent;
using System.Text.Json;
using System.Text.RegularExpressions;
using Framework.Dashboards;
using Framework.Documents;
using Framework.Security;

namespace Framework.Api;

public static class QueryApiOutputs
{
    public const string Rows = "rows", Object = "object", Csv = "csv";
    public static readonly IReadOnlySet<string> All = new HashSet<string> { Rows, Object, Csv };
}

/// <summary>Konfigurasi API dari query tersimpan (JSON). Jalur yang sama dengan widget dashboard & template laporan.</summary>
public sealed class QueryApiConfig
{
    public int SchemaVersion { get; set; } = 1;
    public string Sql { get; set; } = "";
    public List<WidgetParameter> Parameters { get; set; } = [];
    /// <summary>rows = daftar baris JSON · object = baris pertama sebagai objek (KPI) · csv = file CSV.</summary>
    public string Output { get; set; } = QueryApiOutputs.Rows;
    public int MaxRows { get; set; } = 1000;
    /// <summary>0 = tanpa cache. Cache per tenant + nilai parameter (termasuk parameter dari akun).</summary>
    public int CacheSeconds { get; set; }
    /// <summary>Role yang boleh memanggil. Kosong = semua pengguna tenant.</summary>
    public List<string> Roles { get; set; } = [];
    public bool Enabled { get; set; } = true;
}

public sealed class QueryApiDef
{
    public string Id { get; set; } = "";
    public string TenantId { get; set; } = "";
    /// <summary>Bagian URL: GET /api/query/{Slug}. Huruf kecil, angka, tanda hubung.</summary>
    public string Slug { get; set; } = "";
    public string Name { get; set; } = "";
    public string? Description { get; set; }
    public QueryApiConfig Config { get; set; } = new();
    /// <summary>Asal bila diterbitkan dari wizard lain, mis. "widget:w_123" atau "report:rt_456"; null = dibuat langsung.</summary>
    public string? SourceRef { get; set; }
    public string CreatedBy { get; set; } = "";
    public DateTimeOffset CreatedAt { get; set; }
    public DateTimeOffset UpdatedAt { get; set; }
    public int Version { get; set; } = 1;
}

public interface IQueryApiStore
{
    Task<IReadOnlyList<QueryApiDef>> ListAsync(string tenantId, CancellationToken ct = default);
    Task<QueryApiDef?> GetAsync(string tenantId, string id, CancellationToken ct = default);
    Task<QueryApiDef?> GetBySlugAsync(string tenantId, string slug, CancellationToken ct = default);
    Task SaveAsync(QueryApiDef def, CancellationToken ct = default);
    Task DeleteAsync(string tenantId, string id, CancellationToken ct = default);
}

public sealed class InMemoryQueryApiStore : IQueryApiStore
{
    private readonly ConcurrentDictionary<(string, string), string> _items = new();
    private static QueryApiDef Read(string j) => JsonSerializer.Deserialize<QueryApiDef>(j, DashboardJson.Options)!;
    private IEnumerable<QueryApiDef> Of(string t) => _items.Where(kv => kv.Key.Item1 == t).Select(kv => Read(kv.Value));
    public Task<IReadOnlyList<QueryApiDef>> ListAsync(string tenantId, CancellationToken ct = default) => Task.FromResult<IReadOnlyList<QueryApiDef>>(Of(tenantId).OrderBy(d => d.Slug).ToList());
    public Task<QueryApiDef?> GetAsync(string tenantId, string id, CancellationToken ct = default) => Task.FromResult(_items.TryGetValue((tenantId, id), out var j) ? Read(j) : null);
    public Task<QueryApiDef?> GetBySlugAsync(string tenantId, string slug, CancellationToken ct = default) => Task.FromResult(Of(tenantId).FirstOrDefault(d => d.Slug == slug));
    public Task SaveAsync(QueryApiDef def, CancellationToken ct = default) { _items[(def.TenantId, def.Id)] = JsonSerializer.Serialize(def, DashboardJson.Options); return Task.CompletedTask; }
    public Task DeleteAsync(string tenantId, string id, CancellationToken ct = default) { _items.TryRemove((tenantId, id), out _); return Task.CompletedTask; }
}

public sealed record QueryApiResult(QueryApiDef Def, QueryResultSet Data, bool FromCache);

/// <summary>
/// API dari query tersimpan: GET /api/query/{slug}?Param=nilai. Memakai eksekutor read-only dashboard
/// (tenant terisolasi, READ ONLY, statement_timeout, parameter native). Bisa dibuat langsung atau diterbitkan
/// sebagai opsi dari wizard widget dashboard / template laporan (satu jalur query yang sama).
/// </summary>
public sealed partial class QueryApiService(IQueryApiStore store, IReportingQueryExecutor executor, TimeProvider clock)
{
    public const int MaxRowsCap = 10_000, PreviewRows = 100;
    [GeneratedRegex("^[a-z0-9][a-z0-9-]{1,62}$")] private static partial Regex SlugRx();
    private readonly ConcurrentDictionary<string, (DateTimeOffset Expires, QueryResultSet Data)> _cache = new();

    public bool IsAvailable => executor.IsAvailable;
    public static bool CanDesign(UserContext u) => u.Roles.Contains(UserContext.SystemManager);
    public static bool CanCall(UserContext u, QueryApiDef d) => CanDesign(u) || d.Config.Roles.Count == 0 || d.Config.Roles.Any(u.Roles.Contains);
    public static bool IsUserBound(WidgetParameter p) => p.Binding is ParamBindings.UserId or ParamBindings.UserDepartment;
    public static string Slugify(string s) => Regex.Replace((s ?? "").ToLowerInvariant(), "[^a-z0-9]+", "-").Trim('-') is { Length: > 1 } x ? x[..Math.Min(x.Length, 60)] : "query";

    private static void DemandDesigner(UserContext u)
    {
        if (!CanDesign(u)) throw new PermissionDeniedException("Menerbitkan API query hanya untuk System Manager");
    }

    public static (string? Error, List<WidgetParameter> Parameters) DetectParameters(string sql, IReadOnlyList<WidgetParameter>? existing)
    {
        var (a, ps) = DashboardService.DetectParameters(sql, existing);
        if (!a.Ok) return (a.Error, []);
        foreach (var p in ps.Where(p => p.Binding is not (ParamBindings.Default or ParamBindings.UserId or ParamBindings.UserDepartment)))
            p.Binding = ParamBindings.Default; // filter dashboard tidak berlaku: nilai dikirim pemanggil API
        return (null, ps);
    }

    /// <summary>Nilai parameter: dari akun (tidak bisa ditimpa) atau dari pemanggil (default bila tidak dikirim).
    /// Argumen yang tidak dideklarasikan ditolak.</summary>
    private static List<(WidgetParameter P, object? V)> Bind(UserContext u, QueryApiConfig c, IReadOnlyDictionary<string, string?> args)
    {
        var errors = new List<FieldError>();
        foreach (var k in args.Keys.Where(k => c.Parameters.All(p => !p.Name.Equals(k, StringComparison.OrdinalIgnoreCase))))
            errors.Add(new FieldError(k, "parameter tidak dikenal"));
        var values = new List<(WidgetParameter, object?)>();
        foreach (var p in c.Parameters)
        {
            var given = args.FirstOrDefault(kv => kv.Key.Equals(p.Name, StringComparison.OrdinalIgnoreCase));
            if (IsUserBound(p) && given.Key is not null) { errors.Add(new FieldError(p.Name, "diisi otomatis dari akun pemanggil dan tidak boleh dikirim")); continue; }
            var raw = p.Binding switch
            {
                ParamBindings.UserId => u.Id,
                ParamBindings.UserDepartment => u.Attr("department"),
                _ => given.Key is not null ? given.Value : p.DefaultValue,
            };
            try { values.Add((p, DashboardService.Convert(p, raw))); }
            catch (QueryRejectedException ex) { errors.Add(new FieldError(p.Name, ex.Message)); }
        }
        if (errors.Count > 0) throw new ValidationFailedException(errors);
        return values;
    }

    private async Task<QueryResultSet> RunAsync(UserContext u, QueryApiConfig c, List<(WidgetParameter P, object? V)> values, int limit, CancellationToken ct)
    {
        var a = SqlGuard.Analyze(c.Sql);
        if (!a.Ok) throw new QueryRejectedException(a.Error!);
        var ps = a.Parameters.Select(n => values.FirstOrDefault(v => v.P.Name.Equals(n, StringComparison.OrdinalIgnoreCase)) is { P: not null } v
            ? new SqlParam(v.V, v.P.Type) : throw new QueryRejectedException($"Parameter @{n} belum didefinisikan")).ToList();
        return await executor.ExecuteAsync(u.TenantId, a.ParameterizedSql, ps, limit, ct);
    }

    public Task<QueryResultSet> PreviewAsync(UserContext u, QueryApiConfig c, CancellationToken ct = default)
    {
        DemandDesigner(u);
        return RunAsync(u, c, Bind(u, c, new Dictionary<string, string?>()), PreviewRows, ct);
    }

    private async Task Validate(UserContext u, QueryApiDef def, CancellationToken ct)
    {
        if (string.IsNullOrWhiteSpace(def.Name)) throw new ApiMethodException("Nama API wajib diisi");
        if (!SlugRx().IsMatch(def.Slug)) throw new ApiMethodException("Slug harus 2–63 karakter: huruf kecil, angka, atau tanda hubung (mis. rekap-pengajuan)");
        if (!QueryApiOutputs.All.Contains(def.Config.Output)) throw new ApiMethodException("Bentuk keluaran tidak dikenal");
        if (def.Config.MaxRows is < 1 or > MaxRowsCap) throw new ApiMethodException($"Batas baris 1–{MaxRowsCap:N0}");
        if (def.Config.CacheSeconds is < 0 or > 3600) throw new ApiMethodException("Cache 0–3600 detik");
        var (err, _) = DetectParameters(def.Config.Sql, def.Config.Parameters);
        if (err is not null) throw new QueryRejectedException(err);
        foreach (var p in def.Config.Parameters)
            if (p.Binding is not (ParamBindings.Default or ParamBindings.UserId or ParamBindings.UserDepartment)) p.Binding = ParamBindings.Default;
        await RunAsync(u, def.Config, Bind(u, def.Config, new Dictionary<string, string?>()), 1, ct); // query & parameter pasti valid sebelum terbit
    }

    public async Task<QueryApiDef> SaveAsync(UserContext u, QueryApiDef def, CancellationToken ct = default)
    {
        DemandDesigner(u);
        await Validate(u, def, ct);
        var existing = string.IsNullOrEmpty(def.Id) ? null : await store.GetAsync(u.TenantId, def.Id, ct);
        if (await store.GetBySlugAsync(u.TenantId, def.Slug, ct) is { } other && other.Id != existing?.Id)
            throw new ApiMethodException($"Slug '{def.Slug}' sudah dipakai API \"{other.Name}\"");
        var now = clock.GetUtcNow();
        def.TenantId = u.TenantId;
        if (existing is null) { def.Id = "qa_" + Guid.NewGuid().ToString("N")[..12]; def.CreatedBy = u.Id; def.CreatedAt = now; def.Version = 1; }
        else { def.CreatedBy = existing.CreatedBy; def.CreatedAt = existing.CreatedAt; def.Version = existing.Version + 1; def.SourceRef ??= existing.SourceRef; }
        def.UpdatedAt = now;
        await store.SaveAsync(def, ct);
        return def;
    }

    /// <summary>Opsi "terbitkan juga sebagai API" dari wizard widget/laporan: query & parameter disalin dan ikut diperbarui
    /// setiap kali sumbernya disimpan. Pengaturan API (keluaran, batas, cache) dari penerbitan sebelumnya dipertahankan.</summary>
    public async Task<QueryApiDef> PublishFromAsync(UserContext u, string sourceRef, string slug, string name, string? description,
        string sql, IReadOnlyList<WidgetParameter> parameters, IReadOnlyList<string> roles, CancellationToken ct = default)
    {
        DemandDesigner(u);
        var linked = await GetBySourceAsync(u, sourceRef, ct);
        var bySlug = await store.GetBySlugAsync(u.TenantId, slug, ct);
        if (bySlug is not null && bySlug.SourceRef != sourceRef)
            throw new ApiMethodException($"Slug '{slug}' sudah dipakai API \"{bySlug.Name}\"");
        var def = linked ?? new QueryApiDef { SourceRef = sourceRef };
        def.Slug = slug; def.Name = name; def.Description = description;
        def.Config.Sql = sql;
        def.Config.Parameters = JsonSerializer.Deserialize<List<WidgetParameter>>(JsonSerializer.Serialize(parameters, DashboardJson.Options), DashboardJson.Options)!;
        def.Config.Roles = roles.ToList();
        return await SaveAsync(u, def, ct);
    }

    public async Task<QueryApiDef?> GetBySourceAsync(UserContext u, string sourceRef, CancellationToken ct = default) =>
        (await store.ListAsync(u.TenantId, ct)).FirstOrDefault(d => d.SourceRef == sourceRef);

    public async Task<IReadOnlyList<QueryApiDef>> ListAsync(UserContext u, CancellationToken ct = default) =>
        (await store.ListAsync(u.TenantId, ct)).Where(d => CanDesign(u) || d.Config.Enabled && CanCall(u, d)).ToList();

    public async Task<QueryApiDef?> GetAsync(UserContext u, string id, CancellationToken ct = default)
    {
        DemandDesigner(u);
        return await store.GetAsync(u.TenantId, id, ct);
    }

    public async Task DeleteAsync(UserContext u, string id, CancellationToken ct = default)
    {
        DemandDesigner(u);
        await store.DeleteAsync(u.TenantId, id, ct);
    }

    /// <summary>Menjalankan API. Nonaktif / tidak ada → 404 (keberadaan tidak dibocorkan); role → 403; argumen → 422.</summary>
    public async Task<QueryApiResult> ExecuteAsync(UserContext u, string slug, IReadOnlyDictionary<string, string?> args, CancellationToken ct = default)
    {
        var def = await store.GetBySlugAsync(u.TenantId, slug, ct);
        if (def is null || !def.Config.Enabled) throw new KeyNotFoundException($"API query '{slug}' tidak ada");
        if (!CanCall(u, def)) throw new PermissionDeniedException("Anda tidak memiliki akses ke API ini");
        var values = Bind(u, def.Config, args);
        var key = $"{u.TenantId}|{def.Id}|v{def.Version}|" + string.Join("&", values.Select(v => $"{v.P.Name}={JsonSerializer.Serialize(v.V)}"));
        var now = clock.GetUtcNow();
        if (def.Config.CacheSeconds > 0 && _cache.TryGetValue(key, out var hit) && hit.Expires > now)
            return new QueryApiResult(def, hit.Data, true);
        var data = await RunAsync(u, def.Config, values, def.Config.MaxRows, ct);
        if (def.Config.CacheSeconds > 0)
        {
            foreach (var kv in _cache.Where(kv => kv.Value.Expires <= now).ToList()) _cache.TryRemove(kv.Key, out _);
            _cache[key] = (now.AddSeconds(def.Config.CacheSeconds), data);
        }
        return new QueryApiResult(def, data, false);
    }
}

-- DocFrame: API dari query tersimpan (GET /api/query/{slug}). source_ref menautkan API yang diterbitkan dari
-- widget dashboard / template laporan agar ikut diperbarui saat sumbernya disimpan.
CREATE TABLE docframe_query_apis (
    tenant_id   text        NOT NULL,
    id          text        NOT NULL,
    slug        text        NOT NULL,
    name        text        NOT NULL,
    description text,
    config      jsonb       NOT NULL,
    source_ref  text,
    created_by  text        NOT NULL,
    created_at  timestamptz NOT NULL,
    updated_at  timestamptz NOT NULL,
    version     integer     NOT NULL,
    PRIMARY KEY (tenant_id, id),
    UNIQUE (tenant_id, slug)
);

using System.Text.Json;
using Framework.Api;
using Framework.Dashboards;
using Npgsql;

namespace Framework.Persistence.PostgreSql;

public sealed class PostgresQueryApiStore(NpgsqlDataSource db) : IQueryApiStore
{
    private const string Cols = "tenant_id, id, slug, name, description, config::text, source_ref, created_by, created_at, updated_at, version";

    private static QueryApiDef Read(NpgsqlDataReader r) => new()
    {
        TenantId = r.GetString(0), Id = r.GetString(1), Slug = r.GetString(2), Name = r.GetString(3), Description = r.IsDBNull(4) ? null : r.GetString(4),
        Config = JsonSerializer.Deserialize<QueryApiConfig>(r.GetString(5), DashboardJson.Options) ?? new(), SourceRef = r.IsDBNull(6) ? null : r.GetString(6),
        CreatedBy = r.GetString(7), CreatedAt = r.GetFieldValue<DateTimeOffset>(8), UpdatedAt = r.GetFieldValue<DateTimeOffset>(9), Version = r.GetInt32(10),
    };

    private async Task<List<QueryApiDef>> Query(string where, params object[] args)
    {
        await using var cmd = db.CreateCommand($"SELECT {Cols} FROM docframe_query_apis WHERE {where} ORDER BY slug");
        foreach (var a in args) cmd.Parameters.AddWithValue(a);
        var list = new List<QueryApiDef>();
        await using var r = await cmd.ExecuteReaderAsync();
        while (await r.ReadAsync()) list.Add(Read(r));
        return list;
    }

    public async Task<IReadOnlyList<QueryApiDef>> ListAsync(string tenantId, CancellationToken ct = default) => await Query("tenant_id = $1", tenantId);
    public async Task<QueryApiDef?> GetAsync(string tenantId, string id, CancellationToken ct = default) => (await Query("tenant_id = $1 AND id = $2", tenantId, id)).FirstOrDefault();
    public async Task<QueryApiDef?> GetBySlugAsync(string tenantId, string slug, CancellationToken ct = default) => (await Query("tenant_id = $1 AND slug = $2", tenantId, slug)).FirstOrDefault();

    public async Task SaveAsync(QueryApiDef d, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand(
            $"INSERT INTO docframe_query_apis ({Cols.Replace("::text", "")}) VALUES ($1,$2,$3,$4,$5,$6::jsonb,$7,$8,$9,$10,$11) " +
            "ON CONFLICT (tenant_id, id) DO UPDATE SET slug=$3, name=$4, description=$5, config=$6::jsonb, source_ref=$7, updated_at=$10, version=$11");
        cmd.Parameters.AddWithValue(d.TenantId); cmd.Parameters.AddWithValue(d.Id); cmd.Parameters.AddWithValue(d.Slug); cmd.Parameters.AddWithValue(d.Name);
        cmd.Parameters.AddWithValue((object?)d.Description ?? DBNull.Value);
        cmd.Parameters.AddWithValue(JsonSerializer.Serialize(d.Config, DashboardJson.Options));
        cmd.Parameters.AddWithValue((object?)d.SourceRef ?? DBNull.Value);
        cmd.Parameters.AddWithValue(d.CreatedBy); cmd.Parameters.AddWithValue(d.CreatedAt); cmd.Parameters.AddWithValue(d.UpdatedAt); cmd.Parameters.AddWithValue(d.Version);
        await cmd.ExecuteNonQueryAsync(ct);
    }

    public async Task DeleteAsync(string tenantId, string id, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand("DELETE FROM docframe_query_apis WHERE tenant_id = $1 AND id = $2");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(id);
        await cmd.ExecuteNonQueryAsync(ct);
    }
}

using Microsoft.AspNetCore.Routing;

namespace Framework.Web;

public static class DocApi
{
    public static IEndpointRouteBuilder MapDocFrameworkApi(this IEndpointRouteBuilder app)
    {
        // ---------- File (gambar) ----------
        // Cookie browser diterima (dipakai <img>). Tertaut ke dokumen → hanya yang boleh membaca dokumen itu dan
        // dokumen masih memakai file tersebut; belum tertaut → hanya pengunggah. Selain itu 404 (tidak membocorkan keberadaan).
        app.MapGet("/files/{id}", async (HttpContext http, string id) =>
        {
            var user = ClaimsUserMapper.ToUserContext(http.User);
            if (user is null) return Results.Unauthorized();
            var sp = http.RequestServices;
            var file = await sp.GetRequiredService<IFileStore>().GetAsync(user.TenantId, id, http.RequestAborted);
            if (file is not { } f) return Results.NotFound();
            bool allowed;
            if (f.Record.DocName is null) allowed = f.Record.UploadedBy == user.Id;
            else
            {
                try
                {
                    var doc = await Docs(sp).GetAsync(user, f.Record.DocType!, f.Record.DocName, http.RequestAborted);
                    allowed = f.Record.Field is { } field && doc.Data.GetValueOrDefault(field) as string == id;
                }
                catch (Exception ex) when (ex is DocumentNotFoundException or PermissionDeniedException or KeyNotFoundException) { allowed = false; }
            }

    private static int IfMatch(HttpContext http)
    {
        var raw = http.Request.Headers.IfMatch.FirstOrDefault()?.Trim('"', ' ');

    private static Dictionary<string, object?> ToPayload(Dictionary<string, JsonElement> body) =>

    private static List<FilterClause> ParseFilters(string? json)
    {
        if (string.IsNullOrWhiteSpace(json)) return [];
        try
        {
            using var doc = JsonDocument.Parse(json);
            return doc.RootElement.EnumerateArray().Select(e =>
            {
                var a = e.EnumerateArray().ToArray();
                if (a.Length != 3) throw new InvalidPayloadException("Filter harus berupa [field, operator, nilai]");
                object? value = a[2].ValueKind == JsonValueKind.Array
                    ? a[2].EnumerateArray().Select(Scalar).ToList()
                    : Scalar(a[2]);
                return new FilterClause(a[0].GetString()!, a[1].GetString()!, value);
            }).ToList();
        }
        catch (JsonException) { throw new InvalidPayloadException("Parameter filters bukan JSON yang valid"); }
    }

    private static object? Scalar(JsonElement e) => e.ValueKind switch
    {
        JsonValueKind.String => e.GetString(),
        JsonValueKind.Number => e.GetDecimal(),
        JsonValueKind.True => true,
        JsonValueKind.False => false,
        JsonValueKind.Null => null,
        _ => throw new InvalidPayloadException("Nilai filter harus skalar atau daftar skalar")
    };

    public static Dictionary<string, object?> ToDto(Document d)
    {

            ["version"] = d.Version, ["schema_version"] = d.SchemaVersion, ["workflow_version"] = d.WorkflowVersion,

        };

        return dto;
    }

    /// <summary>Pemetaan exception → response yang konsisten (application/problem+json).</summary>
    /// <summary>Endpoint tanpa deklarasi cakupan: kunci API ditolak (gagal tertutup).</summary>
    private static Task<IResult> Run(HttpContext http, Func<UserContext, IServiceProvider, Task<IResult>> action) => Run(http, null, action);

    /// <summary>Cakupan khusus: kunci API boleh masuk tanpa dicek di sini (endpoint menyaring sendiri, mis. daftar method).</summary>
    private const string AnyApiKeyScope = "*any";

    private static async Task<IResult> Run(HttpContext http, string? scope, Func<UserContext, IServiceProvider, Task<IResult>> action)
    {
        // Kunci API: hanya endpoint yang mendeklarasikan cakupan, dan cakupan itu harus diberikan pada kunci.
        if (http.User.FindFirst(Framework.Web.Identity.ApiKeyDefaults.KeyIdClaim) is not null && scope != AnyApiKeyScope)
        {
            if (scope is null) return Problem(403, "Kunci API tidak berlaku", "Endpoint ini tidak bisa dipakai dengan kunci API.");
            if (!ApiScopes.Allows(http.User.FindAll(Framework.Web.Identity.ApiKeyDefaults.ScopeClaim).Select(c => c.Value), scope))
                return Problem(403, "Di luar cakupan kunci API", $"Kunci ini tidak mencakup '{scope}'.");
        }
        // Identitas dari klaim (cookie atau bearer) yang sudah divalidasi middleware autentikasi.
        var user = ClaimsUserMapper.ToUserContext(http.User);
        if (user is null) return Problem(401, "Belum terautentikasi", "Login lewat POST /api/auth/login lalu kirim header Authorization: Bearer <token>.");
        // Cookie dikirim otomatis oleh browser → permintaan yang mengubah data wajib memakai bearer token (perlindungan CSRF).
        var authz = http.Request.Headers.Authorization.ToString();
        var viaBearer = authz.StartsWith("Bearer ", StringComparison.OrdinalIgnoreCase) || authz.StartsWith(Framework.Web.Identity.ApiKeyDefaults.Scheme + " ", StringComparison.OrdinalIgnoreCase);
        if (!viaBearer && !HttpMethods.IsGet(http.Request.Method) && !HttpMethods.IsHead(http.Request.Method))
            return Problem(403, "Butuh bearer token", "API yang mengubah data harus memakai header Authorization: Bearer <token>, bukan cookie browser.");
        try { return await action(user, http.RequestServices); }
        catch (ValidationFailedException ex) { return Problem(422, "Validasi gagal", ex.Message, ex.Errors); }
        catch (PermissionDeniedException ex) { return Problem(403, "Akses ditolak", ex.Message); }
        catch (DocumentNotFoundException ex) { return Problem(404, "Tidak ditemukan", ex.Message); }
        catch (KeyNotFoundException ex) { return Problem(404, "Tidak ditemukan", ex.Message); }
        catch (ConcurrencyException ex) { return Problem(409, "Konflik versi", ex.Message); }
        catch (InvalidStateException ex) { return Problem(409, "Status dokumen tidak mengizinkan", ex.Message); }
        catch (InvalidPayloadException ex) { return Problem(400, "Payload tidak valid", ex.Message, ex.Fields.Select(f => new FieldError(f, "tidak diizinkan")).ToList()); }
        catch (MetadataCompilationException ex) { return Problem(422, "Metadata tidak valid", "Periksa daftar error", ex.Errors.Select(e => new FieldError(e.Path, e.Message)).ToList()); }
        catch (JsonException ex) { return Problem(400, "JSON tidak valid", ex.Message); }
        catch (PdfUnavailableException ex) { return Problem(501, "PDF tidak tersedia", ex.Message); }
        catch (InvalidOperationException ex) { return Problem(409, "Operasi ditolak", ex.Message); }
        catch (ApiMethodException ex) { return Problem(400, "Permintaan ditolak", ex.Message); }
        catch (Framework.Dashboards.QueryRejectedException ex) { return Problem(400, "Query ditolak", ex.Message); }
        catch (Framework.Dashboards.ReportingUnavailableException ex) { return Problem(503, "Fitur query tidak tersedia", ex.Message); }
        catch (Exception ex) when (ex is not OperationCanceledException)
        {
            // Jangan bocorkan detail internal (stack, SQL) ke klien API; detail ada di log server.
            http.RequestServices.GetRequiredService<ILoggerFactory>().CreateLogger("Framework.Api").LogError(ex, "API {Method} {Path} gagal", http.Request.Method, http.Request.Path);
            return Problem(500, "Kesalahan internal", "Terjadi kesalahan di server. Hubungi administrator (rujuk waktu kejadian).");
        }
    }

    /// <summary>CSV UTF-8 (BOM agar Excel membaca huruf dengan benar). Nilai berawalan = + - @ diberi apostrof:
    /// mencegah formula injection saat file dibuka di spreadsheet.</summary>
    private static byte[] Csv(IReadOnlyList<Framework.Dashboards.ResultColumn> cols, IReadOnlyList<object?[]> rows)
    {
        static string Cell(object? v)
        {
            var s = v switch { null => "", decimal d => d.ToString(System.Globalization.CultureInfo.InvariantCulture), bool b => b ? "true" : "false", _ => Convert.ToString(v, System.Globalization.CultureInfo.InvariantCulture) ?? "" };
            if (s.Length > 0 && v is string && "=+-@".Contains(s[0])) s = "'" + s;
            return s.IndexOfAny([',', '"', '\n', '\r']) >= 0 ? "\"" + s.Replace("\"", "\"\"") + "\"" : s;
        }
        var sb = new System.Text.StringBuilder();
        sb.AppendLine(string.Join(",", cols.Select(c => Cell(c.Name))));
        foreach (var r in rows) sb.AppendLine(string.Join(",", r.Select(Cell)));
        return [.. System.Text.Encoding.UTF8.GetPreamble(), .. System.Text.Encoding.UTF8.GetBytes(sb.ToString())];
    }

    private static IResult Problem(int status, string title, string detail, IReadOnlyList<FieldError>? errors = null) =>
        Results.Json(new { type = $"https://httpstatuses.io/{status}", title, status, detail, errors = errors ?? [] },
            statusCode: status, contentType: "application/problem+json");
}

using Framework.Api;
using Framework.Dashboards;
using Framework.Security;

namespace Framework.Web.QueryApis;

/// <summary>
/// Opsi "Terbitkan juga sebagai API" di wizard widget dashboard & template laporan: query dan parameter yang sama
/// diterbitkan di /api/query/{slug} dan ikut diperbarui setiap kali sumbernya disimpan. Dicentang lalu dilepas = API dihapus.
/// </summary>
public sealed class ApiPublishOption
{
    public bool On { get; set; }
    public string Slug { get; set; } = "";
    public QueryApiDef? Linked { get; private set; }

    public async Task LoadAsync(QueryApiService apis, UserContext u, string? sourceRef)
    {
        if (sourceRef is null || !apis.IsAvailable || !QueryApiService.CanDesign(u)) return;
        Linked = await apis.GetBySourceAsync(u, sourceRef);
        On = Linked is not null;
        Slug = Linked?.Slug ?? "";
    }

    /// <summary>Menerapkan pilihan setelah sumber disimpan. Mengembalikan keterangan untuk notifikasi (null = tidak ada perubahan).</summary>
    public async Task<string?> ApplyAsync(QueryApiService apis, UserContext u, string sourceRef, string name, string? description,
        string sql, IReadOnlyList<WidgetParameter> parameters, IReadOnlyList<string> roles)
    {
        if (On)
        {
            var d = await apis.PublishFromAsync(u, sourceRef, Slug, name, description, sql, parameters, roles);
            Linked = d;
            return $"API /api/query/{d.Slug} diperbarui (v{d.Version}).";
        }
        if (Linked is null) return null;
        await apis.DeleteAsync(u, Linked.Id);
        Linked = null;
        return "API dari sumber ini dihapus.";
    }
}

    [Parameter] public string? ApiId { get; set; }

    private static readonly (int, string)[] Steps = [(1, "Query"), (2, "Parameter & pratinjau"), (3, "Terbitkan")];
    private static readonly (string, string, string)[] Outputs =
    [
        (QueryApiOutputs.Rows, "Daftar baris", "{ data: [ {…}, … ], columns, count, truncated }"),
        (QueryApiOutputs.Object, "Satu objek", "baris pertama, cocok untuk KPI: { data: {…} }"),
        (QueryApiOutputs.Csv, "CSV", "file text/csv untuk spreadsheet / ETL"),
    ];
    private static readonly IEnumerable<SelectOption<string>> TypeOptions =
        [new(ParamTypes.Text, "Teks"), new(ParamTypes.Number, "Angka"), new(ParamTypes.Date, "Tanggal"), new(ParamTypes.Boolean, "Boolean")];

    private QueryApiDef def = new();
    private int step = 1, maxStep = 1;
    private bool isEdit, busy, slugTouched;
    private string? error, errorTitle, loadedKey;
    private QueryResultSet? preview;

    private static IEnumerable<SelectOption<string>> BindingOptions(WidgetParameter p)
    {
        yield return new(ParamBindings.Default, "Dari pemanggil");
        if (p.Type == ParamTypes.Text) { yield return new(ParamBindings.UserDepartment, "Dari akun: departemen"); yield return new(ParamBindings.UserId, "Dari akun: ID pengguna"); }
    }

    protected override async Task OnParametersSetAsync()
    {
        if (ApiId == loadedKey && loadedKey is not null) return;
        loadedKey = ApiId ?? "";
        if (ApiId is null || !QueryApiService.CanDesign(Current.Required)) return;
        if (await Apis.GetAsync(Current.Required, ApiId) is not { } d) { (errorTitle, error) = ("Tidak ditemukan", "API query tidak ditemukan."); return; }
        def = d; isEdit = true; slugTouched = true; maxStep = 2;
    }

    private void GoTo(int n) { if (n <= maxStep) { step = n; error = null; } }
    private void Insert(string text) { var s = def.Config.Sql.TrimEnd(); def.Config.Sql = s + (s.Length == 0 ? "" : " ") + text; }
    private void SetName(string? v) { def.Name = v ?? string.Empty; if (!slugTouched) def.Slug = QueryApiService.Slugify(def.Name); }

    private void Detect()
    {
        error = null;
        var (err, ps) = QueryApiService.DetectParameters(def.Config.Sql, def.Config.Parameters);
        if (err is not null) { (errorTitle, error) = ("Query belum bisa dipakai", err); return; }
        def.Config.Parameters = ps; preview = null;
        step = maxStep = 2;
    }

    private async Task RunPreview()
    {
        error = null; busy = true;
        try { preview = await Apis.PreviewAsync(Current.Required, def.Config); maxStep = 3; }
        catch (Exception ex) when (ex is QueryRejectedException or ReportingUnavailableException or ValidationFailedException)
        {
            preview = null; maxStep = 2;
            (errorTitle, error) = ("Query gagal dijalankan", ex is ValidationFailedException v ? string.Join("; ", v.Errors.Select(e => $"{e.Path}: {e.Message}")) : ex.Message);
        }
        finally { busy = false; }
    }

    private string ExampleQuery
    {
        get
        {
            var ps = def.Config.Parameters.Where(p => !QueryApiService.IsUserBound(p)).Select(p => $"{p.Name}={Uri.EscapeDataString(p.DefaultValue ?? (p.Type == ParamTypes.Date ? "2026-01-31" : p.Type == ParamTypes.Number ? "0" : "nilai"))}");
            return ps.Any() ? "?" + string.Join("&", ps) : "";
        }
    }

    private string ExampleResponse
    {
        get
        {
            if (preview is null) return "—";
            Dictionary<string, object?> Row(object?[] r) => preview.Columns.Select((c, i) => (c.Name, r[i])).ToDictionary(x => x.Name, x => x.Item2);
            var opt = new JsonSerializerOptions { WriteIndented = true, Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping };
            return def.Config.Output switch
            {
                QueryApiOutputs.Object => JsonSerializer.Serialize(new { data = preview.Rows.Count > 0 ? Row(preview.Rows[0]) : null }, opt),
                QueryApiOutputs.Csv => string.Join(",", preview.Columns.Select(c => c.Name)) + "\n" + string.Join("\n", preview.Rows.Take(3).Select(r => string.Join(",", r.Select(v => Convert.ToString(v, System.Globalization.CultureInfo.InvariantCulture))))),
                _ => JsonSerializer.Serialize(new { data = preview.Rows.Take(2).Select(Row), count = preview.Rows.Count, truncated = false }, opt),
            };
        }
    }

    private async Task Save()
    {
        error = null; busy = true;
        try
        {
            var saved = await Apis.SaveAsync(Current.Required, def);
            Toasts.Success($"GET /api/query/{saved.Slug}", $"{saved.Name} v{saved.Version} diterbitkan");
            Nav.NavigateTo("/query-apis");
        }
        catch (Exception ex) when (ex is ApiMethodException or QueryRejectedException or ReportingUnavailableException or PermissionDeniedException or ValidationFailedException)
        {
            (errorTitle, error) = ("API belum bisa diterbitkan", ex is ValidationFailedException v ? string.Join("; ", v.Errors.Select(e => $"{e.Path}: {e.Message}")) : ex.Message);
        }
        finally { busy = false; }
    }

    protected override async Task OnInitializedAsync() => await Reload();

    private async Task Reload()
    {
        if (!Current.Required.Roles.Contains(UserContext.SystemManager)) return;
        keys = await Keys.ListAsync(Current.Required);
        accounts = (await Accounts.ListAsync(Current.Required.TenantId)).ToList();
        names = accounts.ToDictionary(a => a.Id, a => a.FullName);
        if (QueryApis.IsAvailable) queryApis = (await QueryApis.ListAsync(Current.Required)).Where(q => q.Config.Enabled).ToList();
    }

    private void StartCreate()
    {
        creating = true; token = null; error = null; limited = false; scopes.Clear();
        name = ""; accountId = NewAccount; newName = ""; newUser = ""; newDept = null; expires = null; ips = null;
    }

    private void Toggle(string scope, bool on) { if (on) scopes.Add(scope); else scopes.Remove(scope); }

    private async Task Create()
    {
        error = null; busy = true;
        try
        {
            var owner = accountId;
            if (accountId == NewAccount)
            {
                if (string.IsNullOrWhiteSpace(newName) || !System.Text.RegularExpressions.Regex.IsMatch(newUser, "^[a-z0-9._]{3,40}$"))
                { error = "Isi nama akun dan username (3–40 karakter: huruf kecil, angka, titik, garis bawah)."; return; }
                if (newRoles.Count == 0) { error = "Pilih minimal satu role untuk akun integrasi."; return; }
                // Akun integrasi dibuat TANPA password: tidak bisa login lewat layar, hanya lewat kunci API.
                var acc = new UserAccount { UserName = newUser, FullName = newName.Trim(), TenantId = Current.Required.TenantId,
                    Department = string.IsNullOrWhiteSpace(newDept) ? null : newDept.Trim(), Roles = newRoles.ToList() };
                var result = await Users.CreateAsync(acc);
                if (!result.Succeeded) { error = string.Join(" ", result.Errors.Select(e => e.Description)); return; }
                owner = acc.Id;
            }
            if (limited && scopes.Count == 0) { error = "Pilih minimal satu cakupan, atau pilih \"Semua yang diizinkan role akun\"."; return; }
            DateTimeOffset? exp = DateOnly.TryParse(expires, out var d) ? new DateTimeOffset(d.ToDateTime(new TimeOnly(23, 59, 59)), TimeZoneInfo.Local.GetUtcOffset(DateTime.Now)) : null;
            var (key, t) = await Keys.CreateAsync(Current.Required, new CreateApiKeyRequest(name, owner,
                limited ? scopes.ToList() : [], (ips ?? "").Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries), exp));
            token = t; creating = false;
            Toasts.Success("Salin kunci sekarang; setelah halaman ditutup kunci tidak bisa dilihat lagi.", $"Kunci \"{key.Name}\" dibuat");
            await Reload();
        }
        catch (Exception ex) when (ex is ApiKeyException or PermissionDeniedException) { error = ex.Message; }
        finally { busy = false; }
    }

    private async Task Revoke(ApiKey k)
    {
        var ok = await Dialogs.ConfirmAsync($"Cabut kunci \"{k.Name}\"?", "Sistem yang memakai kunci ini langsung ditolak (HTTP 401). Tindakan ini tidak bisa dibatalkan.",
            new ConfirmDialogOptions { Destructive = true, ConfirmText = "Cabut kunci", CancelText = "Batal" });
        if (!ok.Confirmed) return;
        await Keys.RevokeAsync(Current.Required, k.Id);
        Toasts.Success("Permintaan dengan kunci ini kini ditolak.", "Kunci dicabut");
        await Reload();
    }

    private static string OpLabel(string op) => op switch { "list" => "Daftar", "read" => "Baca", "create" => "Buat", "update" => "Ubah", "delete" => "Hapus", _ => "Aksi" };
}
