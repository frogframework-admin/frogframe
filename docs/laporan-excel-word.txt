RINGKASAN
---------
Menu "Laporan Excel/Word" (/report-templates). System Manager membuat template lewat
wizard 4 langkah: query detail (+ query header opsional) → parameter & pratinjau →
template Excel/Word bertag (unduh contoh, unggah, pemeriksaan tag, uji coba) → simpan
(nama, pola nama file, role). Pengguna membuka laporan, mengisi parameter, mengunduh.
API: POST /api/report-templates/{id}/generate (bearer).

     (bawaan .NET). Tanpa lisensi komersial, tanpa dependensi NuGet tambahan.

  7. Template contoh otomatis dari kolom query; template dicek saat disimpan.

using System.Globalization;
using System.IO.Compression;
using System.Text;

namespace Framework.Reports;

// ============================================================================
//  Mesin template Excel (.xlsx) & Word (.docx) tanpa library pihak ketiga.
//  File Office Open XML adalah ZIP berisi XML; diproses langsung dengan System.IO.Compression + System.Xml.Linq.
//  Tag:  {{Nama}}            data tunggal (parameter → query header → baris pertama detail → tag sistem)
//        {{Sum:Kolom}}        agregat detail (Sum, Avg, Min, Max, Count)
//        [[Kolom]] / [[No]]   perulangan: baris Excel / baris tabel Word / paragraf Word diulang per baris data
//        ...|format           format opsional, mis. {{Tanggal|dd MMMM yyyy}}, [[Gaji|N0]]
// ============================================================================

public sealed class TemplateException(string message) : Exception(message);

public static class ReportFormats
{
    public const string Xlsx = "xlsx", Docx = "docx";
    public static string ContentType(string f) => f == Xlsx
        ? "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        : "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
}

public enum TagKind { Scalar, Aggregate, Loop }

public sealed record TemplateTag(TagKind Kind, string Name, string? Column, string? Format, string Raw, string Location);

public static class TemplateTags
{
    public static readonly Regex ScalarRx = new(@"\{\{\s*(?<name>[\p{L}_][\p{L}\p{N}_]*)(?:\s*:\s*(?<col>[\p{L}_][\p{L}\p{N}_]*))?\s*(?:\|(?<fmt>[^}]*))?\}\}", RegexOptions.Compiled);
    public static readonly Regex LoopRx = new(@"\[\[\s*(?<name>[\p{L}_][\p{L}\p{N}_]*)\s*(?:\|(?<fmt>[^\]]*))?\]\]", RegexOptions.Compiled);
    public static readonly string[] Aggregates = ["Sum", "Avg", "Min", "Max", "Count"];
    public static readonly string[] System = ["Today", "Now", "User", "UserId", "Tenant", "RowCount"];
    public const string RowNumber = "No";

    public static IEnumerable<TemplateTag> Find(string text, string location)
    {
        foreach (Match m in ScalarRx.Matches(text))
        {
            var col = m.Groups["col"].Success ? m.Groups["col"].Value : null;
            yield return new TemplateTag(col is null && !Aggregates.Contains(m.Groups["name"].Value, StringComparer.OrdinalIgnoreCase) ? TagKind.Scalar : TagKind.Aggregate,
                m.Groups["name"].Value, col, Fmt(m), m.Value, location);
        }
        foreach (Match m in LoopRx.Matches(text))
            yield return new TemplateTag(TagKind.Loop, m.Groups["name"].Value, null, Fmt(m), m.Value, location);
    }

    private static string? Fmt(Match m) => m.Groups["fmt"].Success && m.Groups["fmt"].Value.Trim().Length > 0 ? m.Groups["fmt"].Value.Trim() : null;

    /// <summary>Mencocokkan tag dengan kolom hasil query & parameter. Mengembalikan daftar masalah (kosong = valid).</summary>
    public static IReadOnlyList<string> Validate(IEnumerable<TemplateTag> tags, IReadOnlyList<ResultColumn> detail,
        IReadOnlyList<ResultColumn>? header, IEnumerable<string> parameters)
    {
        var problems = new List<string>();
        bool Has(IEnumerable<ResultColumn>? cols, string n) => cols?.Any(c => c.Name.Equals(n, StringComparison.OrdinalIgnoreCase)) == true;
        var ps = parameters.ToHashSet(StringComparer.OrdinalIgnoreCase);
        foreach (var t in tags.DistinctBy(t => (t.Kind, t.Name.ToLowerInvariant(), t.Column?.ToLowerInvariant())))
        {
            switch (t.Kind)
            {
                case TagKind.Scalar when !ps.Contains(t.Name) && !Has(header, t.Name) && !Has(detail, t.Name) && !System.Contains(t.Name, StringComparer.OrdinalIgnoreCase):
                    problems.Add($"{t.Raw} ({t.Location}): tidak ada parameter, kolom query, atau tag sistem bernama '{t.Name}'");
                    break;
                case TagKind.Aggregate when !Aggregates.Contains(t.Name, StringComparer.OrdinalIgnoreCase):
                    problems.Add($"{t.Raw} ({t.Location}): fungsi '{t.Name}' tidak dikenal (pakai {string.Join(", ", Aggregates)})");
                    break;
                case TagKind.Aggregate when t.Column is not null && !Has(detail, t.Column):
                    problems.Add($"{t.Raw} ({t.Location}): kolom '{t.Column}' tidak ada di hasil query detail");
                    break;
                case TagKind.Aggregate when !t.Name.Equals("Count", StringComparison.OrdinalIgnoreCase)
                                            && (t.Column is null || detail.First(c => c.Name.Equals(t.Column, StringComparison.OrdinalIgnoreCase)).Kind != "number"):
                    problems.Add($"{t.Raw} ({t.Location}): {t.Name} butuh kolom angka");
                    break;
                case TagKind.Loop when !Has(detail, t.Name) && !t.Name.Equals(RowNumber, StringComparison.OrdinalIgnoreCase):
                    problems.Add($"{t.Raw} ({t.Location}): kolom '{t.Name}' tidak ada di hasil query detail");
                    break;
            }
        }
        return problems;
    }
}

/// <summary>Data untuk mengisi template. Kolom dicocokkan tanpa membedakan huruf besar/kecil.</summary>
public sealed class TemplateData
{
    public required QueryResultSet Detail { get; init; }
    public QueryResultSet? Header { get; init; }
    /// <summary>Nilai parameter yang sudah bertipe (string, decimal, DateOnly, bool, null) beserta tipe parameternya.</summary>
    public IReadOnlyDictionary<string, (object? Value, string Type)> Parameters { get; init; } = new Dictionary<string, (object?, string)>();
    public string UserName { get; init; } = "";
    public string UserId { get; init; } = "";
    public string TenantId { get; init; } = "";
    public DateTimeOffset Now { get; init; } = DateTimeOffset.Now;

    private static int Col(QueryResultSet? r, string name) =>
        r is null ? -1 : r.Columns.Select((c, i) => (c, i)).FirstOrDefault(x => x.c.Name.Equals(name, StringComparison.OrdinalIgnoreCase), (null!, -1)).Item2;

    /// <summary>Nilai + jenis (number | date | text | boolean) untuk tag tunggal/agregat.</summary>
    public (object? Value, string Kind) Scalar(TemplateTag t)
    {
        if (t.Kind == TagKind.Aggregate) return Aggregate(t);
        var p = Parameters.FirstOrDefault(kv => kv.Key.Equals(t.Name, StringComparison.OrdinalIgnoreCase));
        if (p.Key is not null)
            return (p.Value.Value, p.Value.Type switch { ParamTypes.Number => "number", ParamTypes.Date => "date", ParamTypes.Boolean => "boolean", _ => "text" });
        foreach (var src in new[] { Header, Detail })
        {
            var i = Col(src, t.Name);
            if (i >= 0) return (src!.Rows.Count > 0 ? src.Rows[0][i] : null, src.Columns[i].Kind);
        }
        return t.Name.ToLowerInvariant() switch
        {
            "today" => (DateOnly.FromDateTime(Now.DateTime), "date"),
            "now" => (Now.DateTime, "date"),
            "user" => (UserName, "text"),
            "userid" => (UserId, "text"),
            "tenant" => (TenantId, "text"),
            "rowcount" => ((decimal)Detail.Rows.Count, "number"),
            _ => throw new TemplateException($"{t.Raw}: nilai tidak ditemukan"),
        };
    }

    private (object? Value, string Kind) Aggregate(TemplateTag t)
    {
        var fn = t.Name.ToLowerInvariant();
        if (fn == "count") return ((decimal)Detail.Rows.Count, "number");
        var i = Col(Detail, t.Column ?? "");
        if (i < 0) throw new TemplateException($"{t.Raw}: kolom '{t.Column}' tidak ada di hasil query");
        var nums = Detail.Rows.Select(r => r[i]).OfType<decimal>().ToList();
        if (nums.Count == 0) return (fn == "sum" ? 0m : null, "number");
        return (fn switch { "sum" => nums.Sum(), "avg" => nums.Average(), "min" => nums.Min(), "max" => nums.Max(), _ => throw new TemplateException($"{t.Raw}: fungsi tidak dikenal") }, "number");
    }

    public (object? Value, string Kind) Loop(TemplateTag t, int row)
    {
        var i = Col(Detail, t.Name);
        if (i >= 0) return (row < Detail.Rows.Count ? Detail.Rows[row][i] : null, Detail.Columns[i].Kind);
        if (t.Name.Equals(TemplateTags.RowNumber, StringComparison.OrdinalIgnoreCase)) return (row < Detail.Rows.Count ? (decimal)(row + 1) : null, "number");
        throw new TemplateException($"{t.Raw}: kolom '{t.Name}' tidak ada di hasil query");
    }

    // ---------- format teks (Word, dan sel Excel yang berisi teks campuran / format eksplisit) ----------
    private static readonly CultureInfo Id = CultureInfo.GetCultureInfo("id-ID");

    public static DateTime? AsDate(object? v) => v switch
    {
        DateTime d => d,
        DateOnly d => d.ToDateTime(TimeOnly.MinValue),
        DateTimeOffset o => o.DateTime,
        string s when DateTime.TryParseExact(s, ["yyyy-MM-dd", "yyyy-MM-dd HH:mm", "yyyy-MM-ddTHH:mm:ss"], CultureInfo.InvariantCulture, DateTimeStyles.None, out var d) => d,
        _ => null,
    };

    public static string Text(object? v, string kind, string? fmt)
    {
        if (v is null) return "";
        try
        {
            if (kind == "date" && AsDate(v) is { } d)
                return d.ToString(fmt ?? (d.TimeOfDay == TimeSpan.Zero ? "dd MMMM yyyy" : "dd MMMM yyyy HH:mm"), Id);
            return v switch
            {
                decimal n => n.ToString(fmt ?? (n == Math.Round(n) ? "#,0" : "#,0.##"), Id),
                bool b => b ? "Ya" : "Tidak",
                _ => Convert.ToString(v, CultureInfo.InvariantCulture) ?? "",
            };
        }
        catch (FormatException) { throw new TemplateException($"Format '{fmt}' tidak valid untuk nilai {v}"); }
    }
}

public sealed class OoxmlPackage
{
    public const long MaxTemplateBytes = 5 * 1024 * 1024;
    public const int MaxEntries = 2_000;
    public const long MaxUncompressedBytes = 80 * 1024 * 1024;

    private readonly List<(string Name, byte[] Data)> _entries = [];
    public IEnumerable<string> Names => _entries.Select(e => e.Name);

    public static OoxmlPackage Open(byte[] bytes, string expectedFormat)
    {
        if (bytes.Length > MaxTemplateBytes) throw new TemplateException($"Template maksimal {MaxTemplateBytes / 1024 / 1024} MB");
        var pkg = new OoxmlPackage();
        try
        {
            using var zip = new ZipArchive(new MemoryStream(bytes), ZipArchiveMode.Read);
            if (zip.Entries.Count > MaxEntries) throw new TemplateException("Template berisi terlalu banyak bagian");
            long total = 0;
            foreach (var e in zip.Entries)
            {
                if (e.FullName.EndsWith('/')) continue;
                using var s = e.Open();
                using var ms = new MemoryStream();
                var buf = new byte[81920];
                int n;
                // Batas dihitung dari byte yang benar-benar didekompresi (tidak percaya ukuran di header ZIP): pertahanan zip bomb.
                while ((n = s.Read(buf)) > 0)
                {
                    total += n;
                    if (total > MaxUncompressedBytes) throw new TemplateException("Template terlalu besar setelah diekstrak (kemungkinan file rusak atau berbahaya)");
                    ms.Write(buf, 0, n);
                }
                pkg._entries.Add((e.FullName, ms.ToArray()));
            }
        }
        catch (InvalidDataException) { throw new TemplateException("File bukan dokumen Office (.xlsx/.docx) yang valid"); }

        var main = expectedFormat == ReportFormats.Xlsx ? "xl/workbook.xml" : "word/document.xml";
        if (!pkg.Has(main) || !pkg.Has("[Content_Types].xml"))
            throw new TemplateException(expectedFormat == ReportFormats.Xlsx ? "File bukan workbook Excel (.xlsx)" : "File bukan dokumen Word (.docx)");
        if (pkg.Names.Any(n => n.EndsWith("vbaProject.bin", StringComparison.OrdinalIgnoreCase)) ||
            Encoding.UTF8.GetString(pkg.Get("[Content_Types].xml")!).Contains("macroEnabled", StringComparison.OrdinalIgnoreCase))
            throw new TemplateException("Template berisi makro (VBA) dan ditolak. Simpan ulang sebagai .xlsx/.docx biasa.");
        return pkg;
    }

    public bool Has(string name) => _entries.Any(e => e.Name == name);
    public byte[]? Get(string name) => _entries.FirstOrDefault(e => e.Name == name).Data;
    public void Remove(string name) => _entries.RemoveAll(e => e.Name == name);
    public void Set(string name, byte[] data)
    {
        var i = _entries.FindIndex(e => e.Name == name);
        if (i >= 0) _entries[i] = (name, data); else _entries.Add((name, data));
    }

    /// <summary>XML diparse tanpa DTD dan tanpa resolver eksternal (mencegah XXE / entity expansion).</summary>
    public XDocument Xml(string name)
    {
        var data = Get(name) ?? throw new TemplateException($"Bagian {name} tidak ada di template");
        try
        {
            using var r = XmlReader.Create(new MemoryStream(data), new XmlReaderSettings { DtdProcessing = DtdProcessing.Prohibit, XmlResolver = null });
            return XDocument.Load(r, LoadOptions.PreserveWhitespace);
        }
        catch (XmlException ex) { throw new TemplateException($"Template rusak atau tidak aman ({name}): {ex.Message}"); }
    }

    public void SetXml(string name, XDocument doc)
    {
        using var ms = new MemoryStream();
        using (var w = XmlWriter.Create(ms, new XmlWriterSettings { Encoding = new UTF8Encoding(false), Indent = false }))
            doc.Save(w);
        Set(name, ms.ToArray());
    }

    public byte[] Save()
    {
        using var ms = new MemoryStream();
        using (var zip = new ZipArchive(ms, ZipArchiveMode.Create, true))
            foreach (var (name, data) in _entries.OrderBy(e => e.Name == "[Content_Types].xml" ? 0 : 1))
            {
                var e = zip.CreateEntry(name, CompressionLevel.Optimal);
                using var s = e.Open();
                s.Write(data);
            }
        return ms.ToArray();
    }

    public static OoxmlPackage FromParts(IEnumerable<(string Name, string Xml)> parts)
    {
        var p = new OoxmlPackage();
        foreach (var (n, x) in parts) p._entries.Add((n, new UTF8Encoding(false).GetBytes(x)));
        return p;
    }
}

public static class XlsxTemplate
{
    private static readonly XNamespace S = "http://schemas.openxmlformats.org/spreadsheetml/2006/main";
    private static readonly XNamespace R = "http://schemas.openxmlformats.org/officeDocument/2006/relationships";
    private static readonly XNamespace PR = "http://schemas.openxmlformats.org/package/2006/relationships";
    private static readonly XNamespace CT = "http://schemas.openxmlformats.org/package/2006/content-types";
    private static readonly XNamespace XmlNs = XNamespace.Xml;

    private sealed record Sheet(string Name, string Path);

    private static List<Sheet> Sheets(OoxmlPackage pkg)
    {
        var wb = pkg.Xml("xl/workbook.xml");
        var rels = pkg.Xml("xl/_rels/workbook.xml.rels").Root!.Elements(PR + "Relationship").ToDictionary(r => (string)r.Attribute("Id")!, r => (string)r.Attribute("Target")!);
        return wb.Root!.Element(S + "sheets")!.Elements(S + "sheet").Select(s =>
        {
            var target = rels[(string)s.Attribute(R + "id")!];
            return new Sheet((string)s.Attribute("name")!, target.StartsWith('/') ? target.TrimStart('/') : "xl/" + target);
        }).ToList();
    }

    private static List<string> SharedStrings(OoxmlPackage pkg) => !pkg.Has("xl/sharedStrings.xml") ? []
        : pkg.Xml("xl/sharedStrings.xml").Root!.Elements(S + "si")
            .Select(si => string.Concat(si.Descendants(S + "t").Where(t => t.Parent?.Name != S + "rPh").Select(t => t.Value))).ToList();

    private static string? CellText(XElement c, List<string> sst)
    {
        if (c.Element(S + "f") is not null) return null; // tag di dalam rumus tidak didukung
        return (string?)c.Attribute("t") switch
        {
            "s" => int.TryParse(c.Element(S + "v")?.Value, out var i) && i < sst.Count ? sst[i] : null,
            "inlineStr" => string.Concat(c.Element(S + "is")?.Descendants(S + "t").Select(t => t.Value) ?? []),
            "str" => c.Element(S + "v")?.Value,
            _ => null,
        };
    }

    public static IReadOnlyList<TemplateTag> Scan(OoxmlPackage pkg)
    {
        var sst = SharedStrings(pkg);
        var tags = new List<TemplateTag>();
        foreach (var sh in Sheets(pkg))
            foreach (var c in pkg.Xml(sh.Path).Descendants(S + "c"))
                if (CellText(c, sst) is { } text)
                    tags.AddRange(TemplateTags.Find(text, $"{sh.Name}!{(string?)c.Attribute("r")}"));
        return tags;
    }

    // ---------- referensi sel & rumus ----------
    public static int ColIndex(string letters) => letters.Aggregate(0, (a, ch) => a * 26 + (ch - 'A' + 1));
    public static string ColLetters(int idx)
    {
        var s = "";
        while (idx > 0) { var m = (idx - 1) % 26; s = (char)('A' + m) + s; idx = (idx - 1) / 26; }
        return s;
    }
    private static readonly Regex CellRefRx = new(@"^(?<c>[A-Z]{1,3})(?<r>\d+)$");
    private static (int Col, int Row) ParseRef(string r) { var m = CellRefRx.Match(r); return (ColIndex(m.Groups["c"].Value), int.Parse(m.Groups["r"].Value)); }

    private static readonly Regex RefRx = new(
        @"(?<![A-Za-z0-9_.$!'""])(?:(?<sheet>'(?:[^']|'')+'|[A-Za-z_][A-Za-z0-9_.]*)!)?(?<ca>\$?)(?<c1>[A-Z]{1,3})(?<ra>\$?)(?<r1>\d+)(?::(?<cb>\$?)(?<c2>[A-Z]{1,3})(?<rb>\$?)(?<r2>\d+))?(?![A-Za-z0-9_(])",
        RegexOptions.Compiled);

    /// <summary>Menerapkan pemetaan ke referensi A1 dalam rumus, di luar string literal; hanya referensi sheet ini.</summary>
    private static string MapRefs(string formula, string sheet, Func<(int c1, int r1, int c2, int r2, bool ca, bool ra, bool cb, bool rb, bool range), (int c1, int r1, int c2, int r2)> map)
    {
        var parts = Regex.Split(formula, "(\"(?:[^\"]|\"\")*\")");
        for (var i = 0; i < parts.Length; i += 2)
            parts[i] = RefRx.Replace(parts[i], m =>
            {
                if (m.Groups["sheet"].Success)
                {
                    var name = m.Groups["sheet"].Value;
                    if (name.StartsWith('\'')) name = name[1..^1].Replace("''", "'");
                    if (!name.Equals(sheet, StringComparison.OrdinalIgnoreCase)) return m.Value;
                }
                var range = m.Groups["c2"].Success;
                var src = (ColIndex(m.Groups["c1"].Value), int.Parse(m.Groups["r1"].Value),
                           range ? ColIndex(m.Groups["c2"].Value) : 0, range ? int.Parse(m.Groups["r2"].Value) : 0,
                           m.Groups["ca"].Value == "$", m.Groups["ra"].Value == "$", m.Groups["cb"].Value == "$", m.Groups["rb"].Value == "$", range);
                var (c1, r1, c2, r2) = map(src);
                var prefix = m.Groups["sheet"].Success ? m.Groups["sheet"].Value + "!" : "";
                var a = $"{m.Groups["ca"].Value}{ColLetters(Math.Max(1, c1))}{m.Groups["ra"].Value}{Math.Max(1, r1)}";
                return prefix + a + (range ? $":{m.Groups["cb"].Value}{ColLetters(Math.Max(1, c2))}{m.Groups["rb"].Value}{Math.Max(1, r2)}" : "");
            });
        return string.Concat(parts);
    }

    /// <summary>Penyisipan <paramref name="delta"/> baris setelah baris T (aturan Excel): referensi di bawah T bergeser,
    /// rentang yang memuat T melebar.</summary>
    private static string InsertShift(string f, string sheet, int t, int delta) =>
        MapRefs(f, sheet, x => x.range
            ? (x.c1, x.r1 > t ? x.r1 + delta : x.r1, x.c2, x.r2 >= t ? x.r2 + delta : x.r2)
            : (x.c1, x.r1 > t ? x.r1 + delta : x.r1, 0, 0));

    /// <summary>Menyalin rumus ke sel lain (seperti fill-down Excel): referensi relatif ikut bergeser.</summary>
    private static string Translate(string f, string sheet, int dRow, int dCol) =>
        MapRefs(f, sheet, x => (x.ca ? x.c1 : x.c1 + dCol, x.ra ? x.r1 : x.r1 + dRow, x.cb ? x.c2 : x.c2 + dCol, x.rb ? x.r2 : x.r2 + dRow));

    private static string InsertShiftRanges(string sqref, int t, int delta) => string.Join(" ", sqref.Split(' ', StringSplitOptions.RemoveEmptyEntries).Select(r =>
    {
        var ab = r.Split(':');
        var (c1, r1) = ParseRef(ab[0]);
        var (c2, r2) = ab.Length > 1 ? ParseRef(ab[1]) : (c1, r1);
        r1 = r1 > t ? r1 + delta : r1;
        r2 = ab.Length > 1 ? (r2 >= t ? r2 + delta : r2) : (r2 > t ? r2 + delta : r2);
        return ab.Length > 1 ? $"{ColLetters(c1)}{r1}:{ColLetters(c2)}{r2}" : $"{ColLetters(c1)}{r1}";
    }));

    // ---------- gaya tanggal ----------
    private sealed class Styles(OoxmlPackage pkg)
    {
        private XDocument? _doc;
        private readonly Dictionary<int, int> _dateOf = [];
        private XDocument Doc => _doc ??= pkg.Xml("xl/styles.xml");
        public bool Changed { get; private set; }

        private bool IsDateFormat(int numFmtId)
        {
            if (numFmtId is >= 14 and <= 22 or >= 45 and <= 47) return true;
            var custom = Doc.Root!.Element(S + "numFmts")?.Elements(S + "numFmt").FirstOrDefault(n => (int)n.Attribute("numFmtId")! == numFmtId);
            if (custom is null) return false;
            var code = Regex.Replace((string)custom.Attribute("formatCode")!, "\"[^\"]*\"|\\[[^\\]]*\\]", "");
            return code.IndexOfAny(['d', 'y', 'D', 'Y']) >= 0;
        }

        /// <summary>Indeks gaya yang menampilkan tanggal: gaya sel template dipakai bila sudah bertipe tanggal, selain itu disalin dengan format tanggal.</summary>
        public int DateStyle(int s)
        {
            if (!pkg.Has("xl/styles.xml")) return s;
            if (_dateOf.TryGetValue(s, out var cached)) return cached;
            var xfs = Doc.Root!.Element(S + "cellXfs")!;
            var list = xfs.Elements(S + "xf").ToList();
            var xf = s < list.Count ? list[s] : list[0];
            if (IsDateFormat((int?)xf.Attribute("numFmtId") ?? 0)) return _dateOf[s] = s;
            var copy = new XElement(xf);
            copy.SetAttributeValue("numFmtId", 14);
            copy.SetAttributeValue("applyNumberFormat", 1);
            xfs.Add(copy);
            xfs.SetAttributeValue("count", list.Count + 1);
            Changed = true;
            return _dateOf[s] = list.Count;
        }

        public void Save() { if (Changed) pkg.SetXml("xl/styles.xml", Doc); }
    }

    // ---------- render ----------
    public static byte[] Render(OoxmlPackage pkg, TemplateData data)
    {
        var sst = SharedStrings(pkg);
        var styles = new Styles(pkg);
        var wb = pkg.Xml("xl/workbook.xml");
        var defined = wb.Root!.Element(S + "definedNames")?.Elements(S + "definedName").ToList() ?? [];
        foreach (var sh in Sheets(pkg))
        {
            var doc = pkg.Xml(sh.Path);
            var ws = doc.Root!;
            var sheetData = ws.Element(S + "sheetData")!;
            DeShare(sheetData, sh.Name);
            var ops = new List<(int T, int Delta)>();

            while (sheetData.Elements(S + "row").FirstOrDefault(r => r.Elements(S + "c").Any(c => CellText(c, sst)?.Contains("[[") == true)) is { } template)
            {
                var t = (int)template.Attribute("r")!;
                var count = Math.Max(data.Detail.Rows.Count, 1); // tanpa data: satu baris kosong (rumus total tetap valid)
                var delta = count - 1;
                var rowMerges = ws.Element(S + "mergeCells")?.Elements(S + "mergeCell")
                    .Where(m => { var ab = ((string)m.Attribute("ref")!).Split(':'); return ab.Length == 2 && ParseRef(ab[0]).Row == t && ParseRef(ab[1]).Row == t; }).ToList() ?? [];
                if (delta > 0)
                {
                    ops.Add((t, delta));
                    foreach (var row in sheetData.Elements(S + "row").Where(r => (int)r.Attribute("r")! > t).ToList())
                    {
                        var nr = (int)row.Attribute("r")! + delta;
                        row.SetAttributeValue("r", nr);
                        foreach (var c in row.Elements(S + "c")) { var (col, _) = ParseRef((string)c.Attribute("r")!); c.SetAttributeValue("r", ColLetters(col) + nr); }
                    }
                    foreach (var f in sheetData.Descendants(S + "f")) f.Value = InsertShift(f.Value, sh.Name, t, delta);
                    foreach (var m in ws.Element(S + "mergeCells")?.Elements(S + "mergeCell").Except(rowMerges) ?? [])
                        m.SetAttributeValue("ref", InsertShiftRanges((string)m.Attribute("ref")!, t, delta));
                    foreach (var el in ws.Descendants().Where(e => e.Name.LocalName is "conditionalFormatting" or "dataValidation" && e.Attribute("sqref") is not null))
                        el.SetAttributeValue("sqref", InsertShiftRanges((string)el.Attribute("sqref")!, t, delta));
                    foreach (var el in ws.Descendants().Where(e => e.Name.LocalName is "hyperlink" or "autoFilter" && e.Attribute("ref") is not null))
                        el.SetAttributeValue("ref", InsertShiftRanges((string)el.Attribute("ref")!, t, delta));
                }
                var clones = new List<XElement>();
                for (var k = 0; k < count; k++)
                {
                    var row = new XElement(template);
                    row.SetAttributeValue("r", t + k);
                    foreach (var c in row.Elements(S + "c"))
                    {
                        var (col, _) = ParseRef((string)c.Attribute("r")!);
                        c.SetAttributeValue("r", ColLetters(col) + (t + k));
                        if (c.Element(S + "f") is { } f && k > 0) f.Value = Translate(f.Value, sh.Name, k, 0);
                        if (CellText(c, sst) is { } text && text.Contains("[["))
                        {
                            var idx = k < data.Detail.Rows.Count ? k : int.MaxValue;
                            Fill(c, text, TemplateTags.LoopRx, m => data.Loop(Tag(m, TagKind.Loop), idx), styles);
                        }
                    }
                    clones.Add(row);
                }
                template.ReplaceWith(clones);
                foreach (var m in rowMerges)
                {
                    var ab = ((string)m.Attribute("ref")!).Split(':');
                    var (c1, _) = ParseRef(ab[0]); var (c2, _) = ParseRef(ab[1]);
                    for (var k = 1; k < count; k++) m.Parent!.Add(new XElement(S + "mergeCell", new XAttribute("ref", $"{ColLetters(c1)}{t + k}:{ColLetters(c2)}{t + k}")));
                }
                if (ws.Element(S + "mergeCells") is { } mc) mc.SetAttributeValue("count", mc.Elements(S + "mergeCell").Count());
            }

            foreach (var c in sheetData.Descendants(S + "c").ToList())
                if (CellText(c, sst) is { } text && text.Contains("{{"))
                    Fill(c, text, TemplateTags.ScalarRx, m => data.Scalar(Tag(m, TemplateTags.Aggregates.Contains(m.Groups["name"].Value, StringComparer.OrdinalIgnoreCase) ? TagKind.Aggregate : TagKind.Scalar)), styles);

            // Nilai rumus yang tersimpan sudah usang: dihapus, Excel menghitung ulang saat dibuka (fullCalcOnLoad).
            foreach (var c in sheetData.Descendants(S + "c").Where(c => c.Element(S + "f") is not null)) { c.Element(S + "v")?.Remove(); if ((string?)c.Attribute("t") is "str" or "e") c.Attribute("t")!.Remove(); }

            var maxRow = sheetData.Elements(S + "row").Select(r => (int)r.Attribute("r")!).DefaultIfEmpty(1).Max();
            if (ws.Element(S + "dimension") is { } dim && ((string)dim.Attribute("ref")!).Split(':') is { Length: 2 } ab2)
            {
                var (c2, _) = ParseRef(ab2[1]);
                dim.SetAttributeValue("ref", $"{ab2[0]}:{ColLetters(c2)}{maxRow}");
            }
            foreach (var (t, delta) in ops)
            {
                foreach (var dn in defined) dn.Value = InsertShift(dn.Value, sh.Name, t, delta);
                ShiftTables(pkg, sh.Path, t, delta);
            }
            pkg.SetXml(sh.Path, doc);
        }

        var calc = wb.Root!.Element(S + "calcPr");
        if (calc is null) wb.Root!.Add(calc = new XElement(S + "calcPr"));
        calc.SetAttributeValue("fullCalcOnLoad", 1);
        pkg.SetXml("xl/workbook.xml", wb);
        RemoveCalcChain(pkg);
        styles.Save();
        return pkg.Save();
    }

    private static TemplateTag Tag(Match m, TagKind kind) => new(kind, m.Groups["name"].Value,
        m.Groups["col"].Success ? m.Groups["col"].Value : null,
        m.Groups["fmt"].Success && m.Groups["fmt"].Value.Trim().Length > 0 ? m.Groups["fmt"].Value.Trim() : null, m.Value, "");

    /// <summary>Isi sel: bila sel hanya berisi satu tag tanpa format → nilai bertipe (angka/tanggal/boolean tetap bisa dihitung);
    /// selain itu teks hasil penggantian. Teks selalu ditulis sebagai inline string — nilai "=…" tidak pernah menjadi rumus.</summary>
    private static void Fill(XElement c, string text, Regex rx, Func<Match, (object? Value, string Kind)> resolve, Styles styles)
    {
        var whole = rx.Match(text.Trim());
        if (whole.Success && whole.Value.Length == text.Trim().Length && !whole.Groups["fmt"].Success)
        {
            var (v, kind) = resolve(whole);
            SetTyped(c, v, kind, styles);
            return;
        }
        SetString(c, rx.Replace(text, m => { var (v, kind) = resolve(m); return TemplateData.Text(v, kind, m.Groups["fmt"].Success ? m.Groups["fmt"].Value.Trim() : null); }));
    }

    private static void Clear(XElement c)
    {
        c.Elements(S + "v").Remove(); c.Elements(S + "is").Remove(); c.Attribute("t")?.Remove();
    }

    private static void SetString(XElement c, string s)
    {
        Clear(c);
        if (s.Length == 0) return;
        c.SetAttributeValue("t", "inlineStr");
        c.Add(new XElement(S + "is", new XElement(S + "t", new XAttribute(XmlNs + "space", "preserve"), s)));
    }

    private static void SetTyped(XElement c, object? v, string kind, Styles styles)
    {
        Clear(c);
        if (v is null) return;
        if (kind == "date" && TemplateData.AsDate(v) is { } d)
        {
            c.Add(new XElement(S + "v", (d - new DateTime(1899, 12, 30)).TotalDays.ToString(CultureInfo.InvariantCulture)));
            c.SetAttributeValue("s", styles.DateStyle((int?)c.Attribute("s") ?? 0));
            return;
        }
        switch (v)
        {
            case decimal n: c.Add(new XElement(S + "v", n.ToString(CultureInfo.InvariantCulture))); break;
            case bool b: c.SetAttributeValue("t", "b"); c.Add(new XElement(S + "v", b ? 1 : 0)); break;
            default: SetString(c, Convert.ToString(v, CultureInfo.InvariantCulture) ?? ""); break;
        }
    }

    /// <summary>Shared formula diubah menjadi rumus biasa per sel agar bisa disalin/digeser dengan benar.</summary>
    private static void DeShare(XElement sheetData, string sheet)
    {
        var masters = new Dictionary<string, (int Col, int Row, string F)>();
        foreach (var c in sheetData.Descendants(S + "c"))
            if (c.Element(S + "f") is { } f && (string?)f.Attribute("t") == "shared" && f.Value.Length > 0)
            {
                var (col, row) = ParseRef((string)c.Attribute("r")!);
                masters[(string)f.Attribute("si")!] = (col, row, f.Value);
            }
        foreach (var c in sheetData.Descendants(S + "c"))
            if (c.Element(S + "f") is { } f && (string?)f.Attribute("t") == "shared")
            {
                if (f.Value.Length == 0 && masters.TryGetValue((string)f.Attribute("si")!, out var m))
                {
                    var (col, row) = ParseRef((string)c.Attribute("r")!);
                    f.Value = Translate(m.F, sheet, row - m.Row, col - m.Col);
                }
                f.Attribute("t")?.Remove(); f.Attribute("si")?.Remove(); f.Attribute("ref")?.Remove();
            }
    }

    /// <summary>Excel Table (ListObject) di sheet ikut melebar agar format & filter tabel mencakup baris baru.</summary>
    private static void ShiftTables(OoxmlPackage pkg, string sheetPath, int t, int delta)
    {
        var relsPath = Path.GetDirectoryName(sheetPath)!.Replace('\\', '/') + "/_rels/" + Path.GetFileName(sheetPath) + ".rels";
        if (!pkg.Has(relsPath)) return;
        foreach (var rel in pkg.Xml(relsPath).Root!.Elements(PR + "Relationship").Where(r => ((string)r.Attribute("Type")!).EndsWith("/table")))
        {
            var target = (string)rel.Attribute("Target")!;
            var path = target.StartsWith('/') ? target.TrimStart('/') : Path.GetFullPath(Path.Combine("/" + Path.GetDirectoryName(sheetPath), target)).TrimStart('/').Replace('\\', '/');
            if (!pkg.Has(path)) continue;
            var tbl = pkg.Xml(path);
            foreach (var el in new[] { tbl.Root! }.Concat(tbl.Root!.Elements(S + "autoFilter")).Where(e => e.Attribute("ref") is not null))
                el.SetAttributeValue("ref", InsertShiftRanges((string)el.Attribute("ref")!, t, delta));
            pkg.SetXml(path, tbl);
        }
    }

    private static void RemoveCalcChain(OoxmlPackage pkg)
    {
        if (!pkg.Has("xl/calcChain.xml")) return;
        pkg.Remove("xl/calcChain.xml");
        var rels = pkg.Xml("xl/_rels/workbook.xml.rels");
        rels.Root!.Elements(PR + "Relationship").Where(r => ((string)r.Attribute("Target")!).EndsWith("calcChain.xml")).Remove();
        pkg.SetXml("xl/_rels/workbook.xml.rels", rels);
        var ct = pkg.Xml("[Content_Types].xml");
        ct.Root!.Elements(CT + "Override").Where(o => ((string)o.Attribute("PartName")!).EndsWith("calcChain.xml")).Remove();
        pkg.SetXml("[Content_Types].xml", ct);
    }

public static class DocxTemplate
{
    private static readonly XNamespace W = "http://schemas.openxmlformats.org/wordprocessingml/2006/main";
    private static readonly XNamespace XmlNs = XNamespace.Xml;
    private static readonly Regex PartRx = new(@"^word/(document|header\d*|footer\d*)\.xml$");

    private static IEnumerable<string> Parts(OoxmlPackage pkg) => pkg.Names.Where(n => PartRx.IsMatch(n)).OrderBy(n => n);

    /// <summary>Elemen teks milik paragraf ini sendiri (bukan paragraf di dalam text box bersarang).</summary>
    private static List<XElement> Texts(XElement p) => p.Descendants(W + "t").Where(t => t.Ancestors(W + "p").First() == p).ToList();
    private static string ParagraphText(XElement p) => string.Concat(Texts(p).Select(t => t.Value));

    public static IReadOnlyList<TemplateTag> Scan(OoxmlPackage pkg)
    {
        var tags = new List<TemplateTag>();
        foreach (var part in Parts(pkg))
            foreach (var p in pkg.Xml(part).Descendants(W + "p"))
                tags.AddRange(TemplateTags.Find(ParagraphText(p), part.Replace("word/", "").Replace(".xml", "")));
        return tags;
    }

    /// <summary>
    /// Penggantian dalam satu paragraf walau tag terpecah ke beberapa run (Word sering memecah "{{Na" + "ma}}" karena
    /// koreksi ejaan / riwayat edit). Teks pengganti ditaruh di run awal tag (format run itu yang dipakai), sisa tag
    /// di run berikutnya dihapus.
    /// </summary>
    private static void ReplaceInParagraph(XElement p, Regex rx, Func<Match, string> value)
    {
        var texts = Texts(p);
        if (texts.Count == 0) return;
        var full = string.Concat(texts.Select(t => t.Value));
        var matches = rx.Matches(full).Cast<Match>().ToList();
        if (matches.Count == 0) return;
        var starts = new int[texts.Count];
        for (int i = 0, pos = 0; i < texts.Count; pos += texts[i].Value.Length, i++) starts[i] = pos;
        int Locate(int index) { for (var i = texts.Count - 1; i >= 0; i--) if (starts[i] <= index && (texts[i].Value.Length > 0 || i == 0)) return i; return 0; }
        foreach (var m in Enumerable.Reverse(matches))
        {
            var replacement = value(m).Replace("\r", "").Replace("\n", " ");
            var a = Locate(m.Index); var b = Locate(m.Index + m.Length - 1);
            var ta = texts[a]; var tb = texts[b];
            var offA = m.Index - starts[a]; var endB = m.Index + m.Length - starts[b];
            if (a == b) ta.Value = ta.Value[..offA] + replacement + ta.Value[(offA + m.Length)..];
            else
            {
                tb.Value = tb.Value[endB..];
                for (var i = a + 1; i < b; i++) texts[i].Value = "";
                ta.Value = ta.Value[..offA] + replacement;
            }
            foreach (var t in new[] { ta, tb }) t.SetAttributeValue(XmlNs + "space", "preserve");
        }
    }

    public static byte[] Render(OoxmlPackage pkg, TemplateData data)
    {
        string Loop(Match m, int row)
        {
            var tag = new TemplateTag(TagKind.Loop, m.Groups["name"].Value, null, m.Groups["fmt"].Success ? m.Groups["fmt"].Value.Trim() : null, m.Value, "");
            var (v, kind) = data.Loop(tag, row);
            return TemplateData.Text(v, kind, tag.Format);
        }
        foreach (var part in Parts(pkg))
        {
            var doc = pkg.Xml(part);
            // 1) Baris tabel berisi [[...]] → diulang per baris data (format baris, border, font ikut tersalin).
            foreach (var tr in doc.Descendants(W + "tr").Where(tr => tr.Descendants(W + "p").Any(p => ParagraphText(p).Contains("[["))).ToList())
                Repeat(tr, data, Loop);
            // 2) Paragraf di luar tabel berisi [[...]] → diulang (mis. daftar berpoin).
            foreach (var p in doc.Descendants(W + "p").Where(p => !p.Ancestors(W + "tbl").Any() && ParagraphText(p).Contains("[[")).ToList())
                Repeat(p, data, Loop);
            // 3) Data tunggal & agregat di seluruh bagian (badan, header, footer).
            foreach (var p in doc.Descendants(W + "p").ToList())
                ReplaceInParagraph(p, TemplateTags.ScalarRx, m =>
                {
                    var col = m.Groups["col"].Success ? m.Groups["col"].Value : null;
                    var kind = col is not null || TemplateTags.Aggregates.Contains(m.Groups["name"].Value, StringComparer.OrdinalIgnoreCase) ? TagKind.Aggregate : TagKind.Scalar;
                    var tag = new TemplateTag(kind, m.Groups["name"].Value, col, m.Groups["fmt"].Success ? m.Groups["fmt"].Value.Trim() : null, m.Value, "");
                    var (v, k) = data.Scalar(tag);
                    return TemplateData.Text(v, k, tag.Format);
                });
            pkg.SetXml(part, doc);
        }
        return pkg.Save();
    }

    private static void Repeat(XElement template, TemplateData data, Func<Match, int, string> loop)
    {
        var clones = new List<XElement>();
        for (var k = 0; k < data.Detail.Rows.Count; k++)
        {
            var clone = new XElement(template);
            var row = k;
            foreach (var p in clone.DescendantsAndSelf(W + "p").ToList()) ReplaceInParagraph(p, TemplateTags.LoopRx, m => loop(m, row));

public sealed class ReportTemplateService(IReportTemplateStore store, IReportingQueryExecutor executor, TimeProvider clock, ReportLimits limits)
{

    public ReportLimits Limits => limits;
    public static bool CanDesign(UserContext u) => u.Roles.Contains(UserContext.SystemManager);
    public static bool CanRun(UserContext u, ReportTemplateDef d) => CanDesign(u) || d.Config.Roles.Count == 0 || d.Config.Roles.Any(u.Roles.Contains);
    private static void DemandDesigner(UserContext u)
    {
        if (!CanDesign(u)) throw new PermissionDeniedException("Merancang template laporan hanya untuk System Manager");
    }

    /// <summary>Parameter dari query detail dan header digabung (nama sama = parameter yang sama).</summary>
    public static (string? Error, List<WidgetParameter> Parameters) DetectParameters(string sql, string? headerSql, IReadOnlyList<WidgetParameter>? existing)
    {
        var (a, ps) = DashboardService.DetectParameters(sql, existing);
        if (!a.Ok) return ($"Query detail: {a.Error}", []);
        if (!string.IsNullOrWhiteSpace(headerSql))
        {
            var (h, hp) = DashboardService.DetectParameters(headerSql, existing);
            if (!h.Ok) return ($"Query header: {h.Error}", []);
            ps.AddRange(hp.Where(p => ps.All(x => !x.Name.Equals(p.Name, StringComparison.OrdinalIgnoreCase))));
        }
        // Binding dashboard (filter global) tidak berlaku di laporan: nilai diisi saat membuat laporan.
        foreach (var p in ps.Where(p => p.Binding is not (ParamBindings.Default or ParamBindings.UserId or ParamBindings.UserDepartment)))
            p.Binding = ParamBindings.Default;
        return (null, ps);
    }

    private static Dictionary<string, (object? Value, string Type)> Values(UserContext u, IReadOnlyList<WidgetParameter> ps, IReadOnlyDictionary<string, string?> input)
    {
        var d = new Dictionary<string, (object?, string)>(StringComparer.OrdinalIgnoreCase);
        foreach (var p in ps)
        {
            var raw = p.Binding switch
            {
                ParamBindings.UserId => u.Id,
                ParamBindings.UserDepartment => u.Attr("department"),
                _ => input.FirstOrDefault(kv => kv.Key.Equals(p.Name, StringComparison.OrdinalIgnoreCase)).Value ?? p.DefaultValue,
            };
            d[p.Name] = (DashboardService.Convert(p, raw), p.Type);
        }
        return d;
    }

    private async Task<QueryResultSet> RunAsync(UserContext u, string sql, Dictionary<string, (object? Value, string Type)> values, int limit, CancellationToken ct)
    {
        var a = SqlGuard.Analyze(sql);
        if (!a.Ok) throw new QueryRejectedException(a.Error!);
        var ps = a.Parameters.Select(n => values.TryGetValue(n, out var v) ? new SqlParam(v.Value, v.Type)
            : throw new QueryRejectedException($"Parameter @{n} belum didefinisikan")).ToList();
        return await executor.ExecuteAsync(u.TenantId, a.ParameterizedSql, ps, limit, ct, limits.StatementTimeout);
    }

    private async Task<ReportPreview> QueryAsync(UserContext u, ReportTemplateConfig c, Dictionary<string, (object?, string)> values, int limit, CancellationToken ct)
    {
        var detail = await RunAsync(u, c.Sql, values, limit, ct);
        var header = string.IsNullOrWhiteSpace(c.HeaderSql) ? null : await RunAsync(u, c.HeaderSql, values, 2, ct);
        if (header is { Rows.Count: > 1 }) throw new QueryRejectedException("Query header harus menghasilkan tepat satu baris");
        return new ReportPreview(detail, header);
    }

    public static TemplateCheck CheckTemplate(ReportTemplateConfig c, byte[] template, ReportPreview preview)
    {
        try
        {
            var pkg = OoxmlPackage.Open(template, c.Format);
            var tags = c.Format == ReportFormats.Xlsx ? XlsxTemplate.Scan(pkg) : DocxTemplate.Scan(pkg);
            var problems = TemplateTags.Validate(tags, preview.Detail.Columns, preview.Header?.Columns, c.Parameters.Select(p => p.Name)).ToList();
            if (tags.Count == 0) problems.Add("Template tidak memuat tag {{…}} atau [[…]] sama sekali");
            return new TemplateCheck(tags, problems);
        }
        catch (TemplateException ex) { return new TemplateCheck([], [ex.Message]); }
    }

    private ReportOutput Render(UserContext u, string name, ReportTemplateConfig c, byte[] template, ReportPreview data, Dictionary<string, (object?, string)> values)
    {
        var td = new TemplateData
        {
            Detail = data.Detail, Header = data.Header, Parameters = values,
            UserName = u.FullName, UserId = u.Id, TenantId = u.TenantId, Now = clock.GetLocalNow(),
        };
        var pkg = OoxmlPackage.Open(template, c.Format);
        var bytes = c.Format == ReportFormats.Xlsx ? XlsxTemplate.Render(pkg, td) : DocxTemplate.Render(pkg, td);
        return new ReportOutput(bytes, FileName(c.FileNamePattern, name, td, c.Format), ReportFormats.ContentType(c.Format), data.Detail.Rows.Count);
    }

    private static string FileName(string pattern, string name, TemplateData td, string format)
    {
        var text = string.IsNullOrWhiteSpace(pattern) ? name : pattern;
        try
        {
            text = TemplateTags.ScalarRx.Replace(text, m =>
            {
                var col = m.Groups["col"].Success ? m.Groups["col"].Value : null;
                var tag = new TemplateTag(col is null && !TemplateTags.Aggregates.Contains(m.Groups["name"].Value, StringComparer.OrdinalIgnoreCase) ? TagKind.Scalar : TagKind.Aggregate,
                    m.Groups["name"].Value, col, m.Groups["fmt"].Success ? m.Groups["fmt"].Value.Trim() : null, m.Value, "");
                var (v, kind) = td.Scalar(tag);
                return TemplateData.Text(v, kind, tag.Format ?? (kind == "date" ? "yyyy-MM-dd" : null));
            });
        }
        catch (TemplateException) { text = name; }
        text = Regex.Replace(text, @"[\\/:*?""<>|\x00-\x1F]+", "-").Trim(' ', '.', '-');
        return (text.Length == 0 ? "laporan" : text[..Math.Min(text.Length, 120)]) + "." + format;
    }

    public async Task<ReportTemplateDef> SaveAsync(UserContext u, ReportTemplateDef def, byte[]? newTemplate, CancellationToken ct = default)
    {
        DemandDesigner(u);
        if (string.IsNullOrWhiteSpace(def.Name)) throw new TemplateException("Nama laporan wajib diisi");
        if (def.Config.Format is not (ReportFormats.Xlsx or ReportFormats.Docx)) throw new TemplateException("Format harus xlsx atau docx");
        var existing = string.IsNullOrEmpty(def.Id) ? null : await store.GetAsync(u.TenantId, def.Id, ct);
        var template = newTemplate ?? (existing is null ? null : await store.GetFileAsync(u.TenantId, def.Id, ct));
        if (template is null || template.Length == 0) throw new TemplateException("Unggah file template terlebih dahulu");
        var (err, _) = DetectParameters(def.Config.Sql, def.Config.HeaderSql, def.Config.Parameters);
        if (err is not null) throw new QueryRejectedException(err);

        // Query dijalankan sekali dan tag dicek terhadap kolomnya: template yang tidak cocok ditolak saat disimpan,
        // bukan baru ketahuan saat pengguna membuat laporan.
        var preview = await QueryAsync(u, def.Config, Values(u, def.Config.Parameters, new Dictionary<string, string?>()), 1, ct);
        var check = CheckTemplate(def.Config, template, preview);
        if (check.Problems.Count > 0) throw new TemplateException("Template belum cocok dengan query: " + string.Join("; ", check.Problems));

        var now = clock.GetUtcNow();
        def.TenantId = u.TenantId;
        if (existing is null) { def.Id = "rt_" + Guid.NewGuid().ToString("N")[..12]; def.CreatedBy = u.Id; def.CreatedAt = now; def.Version = 1; }
        else { def.CreatedBy = existing.CreatedBy; def.CreatedAt = existing.CreatedAt; def.Version = existing.Version + 1; if (newTemplate is null) def.TemplateFileName = existing.TemplateFileName; }
        def.UpdatedAt = now;
        await store.SaveAsync(def, newTemplate, ct);
        return def;
    }

    }

    public async Task<ReportOutput> GenerateAsync(UserContext u, string id, IReadOnlyDictionary<string, string?> input, CancellationToken ct = default)
    {
        var def = await store.GetAsync(u.TenantId, id, ct) ?? throw new TemplateException("Template laporan tidak ditemukan");
        if (!CanRun(u, def)) throw new PermissionDeniedException("Anda tidak memiliki akses ke laporan ini");
        var file = await store.GetFileAsync(u.TenantId, id, ct) ?? throw new TemplateException("File template tidak ditemukan");
        var max = def.Config.Format == ReportFormats.Docx ? limits.MaxRowsWord : limits.MaxRowsExcel;
        var values = Values(u, def.Config.Parameters, input);
        var data = await QueryAsync(u, def.Config, values, max, ct);
        if (data.Detail.Truncated)
            throw new TemplateException($"Hasil melebihi {max:N0} baris. Persempit dengan parameter (mis. rentang tanggal) atau bagi menjadi beberapa laporan.");
        return Render(u, def.Name, def.Config, file, data, values);
    }

    private static ReportTemplateDef Read(NpgsqlDataReader r) => new()
    {
        TenantId = r.GetString(0), Id = r.GetString(1), Name = r.GetString(2), Description = r.IsDBNull(3) ? null : r.GetString(3),
        Config = JsonSerializer.Deserialize<ReportTemplateConfig>(r.GetString(4), DashboardJson.Options) ?? new(),
        TemplateFileName = r.GetString(5), CreatedBy = r.GetString(6), CreatedAt = r.GetFieldValue<DateTimeOffset>(7),

    };

    public async Task<IReadOnlyList<ReportTemplateDef>> ListAsync(string tenantId, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand($"SELECT {Cols} FROM docframe_report_templates WHERE tenant_id = $1 ORDER BY name");
        cmd.Parameters.AddWithValue(tenantId);
        var list = new List<ReportTemplateDef>();
        await using var r = await cmd.ExecuteReaderAsync(ct);
        while (await r.ReadAsync(ct)) list.Add(Read(r));
        return list;
    }

    public async Task<ReportTemplateDef?> GetAsync(string tenantId, string id, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand($"SELECT {Cols} FROM docframe_report_templates WHERE tenant_id = $1 AND id = $2");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(id);
        await using var r = await cmd.ExecuteReaderAsync(ct);
        return await r.ReadAsync(ct) ? Read(r) : null;
    }

        await using var cmd = db.CreateCommand("SELECT template FROM docframe_report_templates WHERE tenant_id = $1 AND id = $2");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(id);
        return await cmd.ExecuteScalarAsync(ct) as byte[];
    }

    public async Task SaveAsync(ReportTemplateDef d, byte[]? file, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand(file is null
            ? "UPDATE docframe_report_templates SET name=$3, description=$4, config=$5::jsonb, updated_at=$6, version=$7 WHERE tenant_id=$1 AND id=$2"
            : "INSERT INTO docframe_report_templates (tenant_id, id, name, description, config, template_file_name, template, created_by, created_at, updated_at, version) " +
              "VALUES ($1,$2,$3,$4,$5::jsonb,$8,$9,$10,$11,$6,$7) ON CONFLICT (tenant_id, id) DO UPDATE SET name=$3, description=$4, config=$5::jsonb, " +
              "template_file_name=$8, template=$9, updated_at=$6, version=$7");

/// <summary>
/// Membuat view reporting.&lt;doctype&gt; (dan reporting.&lt;doctype&gt;_&lt;tabel&gt; untuk child table) dari metadata aktif,
/// berisi kolom standar + field dengan tipe SQL yang sesuai. Field berlevel izin &gt; 0 dan field gambar tidak diekspos.
/// Setiap view memfilter tenant dari docframe_reporting_sessions → query user otomatis terbatas pada tenantnya.
/// </summary>
public sealed class ReportingSchemaSync(AnalyticsDataSource analytics, IMetadataProvider metadata, ReportingOptions options, ILogger<ReportingSchemaSync> log)
{
    private const long LockKey = 0x5265706F7274; // "Report"
    private int _subscribed;

    private static string Ident(string s) => "\"" + s.Replace("\"", "\"\"") + "\"";
    private static string Lit(string s) => "'" + s.Replace("'", "''") + "'";

    /// <summary>
    /// Kolom bertipe dari JSONB. Cast LANGSUNG (tanpa CASE/regex): data hanya ditulis lewat DocumentService yang
    /// menormalkan angka → angka JSON, centang → boolean JSON, tanggal → "yyyy-MM-dd". Pembungkus CASE + regex sebelumnya
    /// mahal per baris dan menyembunyikan filter dari planner — benchmark 1 juta dokumen: agregasi 1.970 ms → 860 ms.
    /// </summary>
    private static string Column(string source, CompiledField f)
    {
        var key = Lit(f.Name);
        return f.Type.ClrType == typeof(long) || f.Type.ClrType == typeof(decimal) ? $"NULLIF({source}->>{key}, '')::numeric"
            : f.Type.ClrType == typeof(bool) ? $"({source}->>{key})::boolean"
            : f.Def.Type == "Date" ? $"NULLIF({source}->>{key}, '')::date"
            : $"({source}->>{key})";
    }

    private const string TenantFilter =
        "d.tenant_id = (SELECT s.tenant_id FROM public.docframe_reporting_sessions s WHERE s.pid = pg_backend_pid())";

    public async Task SyncAsync(CancellationToken ct = default)
    {
        var snap = metadata.Current;
        var ddl = new StringBuilder();
        var views = new List<string>();
        foreach (var meta in snap.DocTypes.Values.Where(m => !m.Def.IsChild))
        {
            var view = ReportingCatalog.SnakeCase(meta.Name);
            views.Add(view);
            var cols = new List<string> { "d.name", "d.owner", "d.docstatus", "d.workflow_state", "d.created", "d.modified" };
            cols.AddRange(meta.Fields.Where(ReportingCatalog.Exposed).Select(f => $"{Column("d.data", f)} AS {Ident(f.Name)}"));
            ddl.AppendLine($"CREATE VIEW reporting.{Ident(view)} AS SELECT {string.Join(", ", cols)} FROM public.docframe_documents d " +
                           $"WHERE d.doctype = {Lit(meta.Name)} AND {TenantFilter};");
            ddl.AppendLine($"COMMENT ON VIEW reporting.{Ident(view)} IS {Lit($"{meta.Title} (DocType {meta.Name} v{meta.Version})")};");
            foreach (var t in meta.TableFields.Where(t => t.Def.PermLevel == 0))
            {
                if (!snap.TryGet(t.Def.ChildDocType!, out var child)) continue;
                var cview = $"{view}_{t.Name}";
                views.Add(cview);
                var key = Lit(t.Name);
                var ccols = new List<string> { "d.name AS parent", "r.ordinality::integer AS idx" };
                ccols.AddRange(child.Fields.Where(ReportingCatalog.Exposed).Select(f => $"{Column("r.value", f)} AS {Ident(f.Name)}"));
                ddl.AppendLine($"CREATE VIEW reporting.{Ident(cview)} AS SELECT {string.Join(", ", ccols)} FROM public.docframe_documents d " +
                               $"CROSS JOIN LATERAL jsonb_array_elements(CASE WHEN jsonb_typeof(d.data->{key}) = 'array' THEN d.data->{key} ELSE '[]'::jsonb END) WITH ORDINALITY AS r(value, ordinality) " +
                               $"WHERE d.doctype = {Lit(meta.Name)} AND {TenantFilter};");
            }
        }

        await using var conn = await analytics.Source.OpenConnectionAsync(ct);
        await using var tx = await conn.BeginTransactionAsync(ct);
        await Exec(conn, tx, $"SELECT pg_advisory_xact_lock({LockKey})", ct);
        await Exec(conn, tx, "CREATE SCHEMA IF NOT EXISTS reporting", ct);
        // Di database aplikasi tabel ini dibuat migrasi 005; di instance analitik terpisah dibuat di sini.
        await Exec(conn, tx, "CREATE TABLE IF NOT EXISTS public.docframe_reporting_sessions (pid integer PRIMARY KEY, tenant_id text NOT NULL, created_at timestamptz NOT NULL)", ct);
        var existing = new List<string>();
        await using (var cmd = new NpgsqlCommand("SELECT viewname FROM pg_views WHERE schemaname = 'reporting'", conn, tx))
        await using (var r = await cmd.ExecuteReaderAsync(ct))
            while (await r.ReadAsync(ct)) existing.Add(r.GetString(0));
        foreach (var v in existing) await Exec(conn, tx, $"DROP VIEW IF EXISTS reporting.{Ident(v)} CASCADE", ct);
        await Exec(conn, tx, ddl.ToString(), ct);

        bool roleExists;
        await using (var cmd = new NpgsqlCommand("SELECT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = $1)", conn, tx))
        {
            cmd.Parameters.AddWithValue(options.ReadOnlyRole);
            roleExists = (bool)(await cmd.ExecuteScalarAsync(ct))!;
        }
        if (roleExists)
        {
            await Exec(conn, tx, $"GRANT USAGE ON SCHEMA reporting TO {Ident(options.ReadOnlyRole)}", ct);
            await Exec(conn, tx, $"GRANT SELECT ON ALL TABLES IN SCHEMA reporting TO {Ident(options.ReadOnlyRole)}", ct);
        }
        else log.LogWarning("Role read-only '{Role}' tidak ditemukan; view reporting dibuat tanpa GRANT", options.ReadOnlyRole);
        await tx.CommitAsync(ct);
        log.LogInformation("Skema reporting disinkronkan: {Count} view", views.Count);

        // Sinkron ulang setiap kali metadata dipublikasikan (DocType/field baru langsung bisa di-query).
        if (Interlocked.Exchange(ref _subscribed, 1) == 0 && metadata is MetadataRegistry reg)
            reg.Published += snapshot => _ = Task.Run(async () =>
            {
                try { await SyncAsync(); } catch (Exception ex) { log.LogError(ex, "Sinkronisasi skema reporting gagal"); }
            });
    }

    private static async Task Exec(NpgsqlConnection c, NpgsqlTransaction tx, string sql, CancellationToken ct)
    {
        await using var cmd = new NpgsqlCommand(sql, c, tx);
        await cmd.ExecuteNonQueryAsync(ct);
    }
}

/// <summary>
/// Menjalankan query widget: koneksi read-only yang diverifikasi, tenant dikunci per sesi, transaksi READ ONLY,
/// statement_timeout, dibungkus LIMIT, parameter dikirim native (bukan digabung ke teks SQL).
/// </summary>
public sealed class PostgresReportingQueryExecutor(AnalyticsDataSource analytics, ReadOnlyDataSource roDb, ReportingOptions options,
    ILogger<PostgresReportingQueryExecutor> log) : IReportingQueryExecutor
{
    private volatile string? _verified;

    public bool IsAvailable => true;

    /// <summary>Menolak connection string yang ternyata bukan read-only (bisa menulis, superuser, atau bisa membaca tabel inti).</summary>
    private async Task VerifyReadOnlyAsync(NpgsqlConnection ro, CancellationToken ct)
    {
        if (_verified == ro.ConnectionString) return;
        // Tabel dicari lewat katalog (OID), bukan nama "public.x": user read-only yang benar memang tidak punya USAGE
        // pada skema public, dan resolusi nama akan gagal. Hak tabel dicek langsung pada OID-nya.
        const string sql = """
            WITH t AS (
                SELECT c.relname, c.oid FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
                WHERE n.nspname = 'public' AND c.relname IN ('docframe_documents', 'docframe_users', 'docframe_reporting_sessions')
            )
            SELECT current_user,
                   r.rolsuper OR r.rolcreaterole OR r.rolcreatedb OR r.rolbypassrls,
                   EXISTS (SELECT 1 FROM information_schema.role_table_grants g
                           WHERE g.grantee = current_user AND g.privilege_type IN ('INSERT','UPDATE','DELETE','TRUNCATE')),
                   COALESCE((SELECT has_table_privilege(current_user, t.oid, 'SELECT') FROM t WHERE t.relname = 'docframe_documents'), false),
                   COALESCE((SELECT has_table_privilege(current_user, t.oid, 'SELECT') FROM t WHERE t.relname = 'docframe_users'), false),
                   COALESCE((SELECT has_table_privilege(current_user, t.oid, 'SELECT,INSERT,UPDATE,DELETE') FROM t WHERE t.relname = 'docframe_reporting_sessions'), false),
                   COALESCE((SELECT has_schema_privilege(current_user, n.oid, 'CREATE') FROM pg_namespace n WHERE n.nspname = 'reporting'), false)
            FROM pg_roles r WHERE r.rolname = current_user
            """;
        await using var cmd = new NpgsqlCommand(sql, ro);
        await using var r = await cmd.ExecuteReaderAsync(ct);
        if (!await r.ReadAsync(ct)) throw new ReportingUnavailableException("Role read-only tidak ditemukan");
        var problems = new List<string>();
        if (r.GetBoolean(1)) problems.Add("role punya hak superuser/create");
        if (r.GetBoolean(2)) problems.Add("role punya hak INSERT/UPDATE/DELETE");
        if (r.GetBoolean(3)) problems.Add("role bisa membaca tabel dokumen langsung (melewati isolasi tenant)");
        if (r.GetBoolean(4)) problems.Add("role bisa membaca tabel akun");
        if (r.GetBoolean(5)) problems.Add("role punya hak atas tabel sesi pelaporan");
        if (r.GetBoolean(6)) problems.Add("role bisa membuat objek di skema reporting");
        var user = r.GetString(0);
        if (problems.Count > 0)
        {
            log.LogError("Connection string read-only ditolak ({User}): {Problems}", user, string.Join("; ", problems));
            throw new ReportingUnavailableException($"Koneksi dashboard bukan read-only yang aman ({user}): {string.Join("; ", problems)}. Gunakan user khusus SELECT pada skema reporting.");
        }
        _verified = ro.ConnectionString;
    }

    public async Task<QueryResultSet> ExecuteAsync(string tenantId, string parameterizedSql, IReadOnlyList<SqlParam> parameters, int limit, CancellationToken ct = default, TimeSpan? timeout = null)
    {
        var sw = Stopwatch.StartNew();
        NpgsqlConnection ro;
        try { ro = await roDb.Source.OpenConnectionAsync(ct); }
        catch (Exception ex) when (ex is NpgsqlException or TimeoutException)
        {
            // Tanpa ini exception mentah lolos ke halaman dashboard dan menghentikannya (kartu tanpa grafik).
            throw new ReportingUnavailableException(ConnectionHint(ex));
        }
        await using var _ro = ro;
        try { await VerifyReadOnlyAsync(ro, ct); }
        catch (PostgresException ex) { throw new ReportingUnavailableException($"Gagal memverifikasi koneksi read-only: {ex.MessageText}"); }
        var pid = ro.ProcessID;
        await using (var reg = analytics.Source.CreateCommand(
            "INSERT INTO docframe_reporting_sessions (pid, tenant_id, created_at) VALUES ($1, $2, now()) ON CONFLICT (pid) DO UPDATE SET tenant_id = $2, created_at = now()"))
        {
            reg.Parameters.AddWithValue(pid); reg.Parameters.AddWithValue(tenantId);
            await reg.ExecuteNonQueryAsync(ct);
        }
        try
        {
            await using var tx = await ro.BeginTransactionAsync(IsolationLevel.RepeatableRead, ct);
            await using (var set = new NpgsqlCommand($"SET TRANSACTION READ ONLY; SET LOCAL statement_timeout = '{Math.Clamp(timeout is { } t ? (int)Math.Ceiling(t.TotalSeconds) : options.StatementTimeoutSeconds, 1, 600)}s'", ro, tx))
                await set.ExecuteNonQueryAsync(ct);

            // Pembungkus LIMIT: query yang salah tulis tidak bisa membawa jutaan baris ke memori aplikasi.
            await using var cmd = new NpgsqlCommand($"SELECT * FROM (\n{parameterizedSql}\n) AS __q LIMIT {limit + 1}", ro, tx);
            foreach (var p in parameters)
                cmd.Parameters.Add(new NpgsqlParameter
                {
                    Value = p.Value ?? DBNull.Value,
                    NpgsqlDbType = p.Type switch
                    {
                        ParamTypes.Number => NpgsqlDbType.Numeric,
                        ParamTypes.Date => NpgsqlDbType.Date,
                        ParamTypes.Boolean => NpgsqlDbType.Boolean,
                        ParamTypes.Text => NpgsqlDbType.Text,
                        _ => throw new QueryRejectedException("Tipe parameter tidak didukung"),
                    },
                });

            var rows = new List<object?[]>();
            var truncated = false;
            List<ResultColumn> columns;
            await using (var r = await cmd.ExecuteReaderAsync(ct))
            {
                columns = Enumerable.Range(0, r.FieldCount).Select(i => new ResultColumn(r.GetName(i), Kind(r.GetFieldType(i)))).ToList();
                while (await r.ReadAsync(ct))
                {
                    if (rows.Count == limit) { truncated = true; break; }
                    var row = new object?[r.FieldCount];
                    for (var i = 0; i < r.FieldCount; i++) row[i] = r.IsDBNull(i) ? null : Normalize(r.GetValue(i));
                    rows.Add(row);
                }
            }
            await tx.RollbackAsync(ct);
            return new QueryResultSet(columns, rows, truncated, sw.ElapsedMilliseconds);
        }
        catch (NpgsqlException ex) when (ex is not PostgresException)
        {
            throw new ReportingUnavailableException($"Koneksi dashboard terputus: {ex.Message}");
        }
        catch (PostgresException ex)
        {
            throw new QueryRejectedException(ex.SqlState switch
            {
                PostgresErrorCodes.QueryCanceled => $"Query melebihi batas waktu {(timeout is { } t ? (int)Math.Ceiling(t.TotalSeconds) : options.StatementTimeoutSeconds)} detik",
                PostgresErrorCodes.InsufficientPrivilege => $"Akses ditolak: {ex.MessageText}. Query hanya boleh membaca view di skema reporting.",
                PostgresErrorCodes.ReadOnlySqlTransaction => "Query mencoba mengubah data; hanya SELECT yang diizinkan.",
                _ => ex.MessageText,
            });
        }
        finally
        {
            await using var del = analytics.Source.CreateCommand("DELETE FROM docframe_reporting_sessions WHERE pid = $1");
            del.Parameters.AddWithValue(pid);
            await del.ExecuteNonQueryAsync(CancellationToken.None);
        }
    }

    /// <summary>Pesan yang bisa ditindaklanjuti untuk kegagalan koneksi read-only (bukan exception mentah).</summary>
    private string ConnectionHint(Exception ex)
    {
        var csb = new NpgsqlConnectionStringBuilder(roDb.Source.ConnectionString);
        var who = $"user '{csb.Username}' ke database '{csb.Database}' di {csb.Host}";
        return ex is PostgresException pg ? pg.SqlState switch
        {
            "28P01" or "28000" => $"Login dashboard gagal untuk {who}. Pastikan role read-only sudah dibuat dengan deploy/sql/reporting-readonly-user.sql " +
                                  "dan password-nya sama dengan ConnectionStrings:DocFrameReadOnly. (Docker compose hanya menjalankan script itu saat volume database baru.)",
            "3D000" => $"Database untuk dashboard tidak ditemukan ({who}). Periksa ConnectionStrings:DocFrameReadOnly.",
            "42501" => $"{who} tidak punya izin CONNECT. Jalankan deploy/sql/reporting-readonly-user.sql (GRANT CONNECT).",
            _ => $"Tidak bisa terhubung sebagai {who}: {pg.MessageText}",
        } : $"Tidak bisa terhubung ke database dashboard sebagai {who}: {ex.Message}";
    }

    private static string Kind(Type t) =>
        t == typeof(decimal) || t == typeof(int) || t == typeof(long) || t == typeof(short) || t == typeof(double) || t == typeof(float) ? "number"
        : t == typeof(DateTime) || t == typeof(DateOnly) || t == typeof(DateTimeOffset) ? "date"
        : t == typeof(bool) ? "boolean" : "text";

    private static object? Normalize(object v) => v switch
    {
        int i => (decimal)i, long l => (decimal)l, short s => (decimal)s, double d => (decimal)d, float f => (decimal)f,
        DateTime dt => dt.TimeOfDay == TimeSpan.Zero ? dt.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture) : dt.ToString("yyyy-MM-dd HH:mm", CultureInfo.InvariantCulture),
        DateOnly d => d.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture),
        DateTimeOffset o => o.ToString("yyyy-MM-dd HH:mm", CultureInfo.InvariantCulture),
        decimal or string or bool => v,
        _ => Convert.ToString(v, CultureInfo.InvariantCulture),
    };
}

public sealed class PostgresWidgetStore(NpgsqlDataSource db) : IWidgetStore
{
    private static WidgetDef Read(NpgsqlDataReader r) => new()
    {
        TenantId = r.GetString(0), Id = r.GetString(1), Name = r.GetString(2), Description = r.IsDBNull(3) ? null : r.GetString(3),
        Config = JsonSerializer.Deserialize<WidgetConfig>(r.GetString(4), DashboardJson.Options) ?? new(),
        CreatedBy = r.GetString(5), CreatedAt = r.GetFieldValue<DateTimeOffset>(6), UpdatedAt = r.GetFieldValue<DateTimeOffset>(7), Version = r.GetInt32(8),
    };
    private const string Cols = "tenant_id, id, name, description, config::text, created_by, created_at, updated_at, version";

    public async Task<IReadOnlyList<WidgetDef>> ListAsync(string tenantId, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand($"SELECT {Cols} FROM docframe_dashboard_widgets WHERE tenant_id = $1 ORDER BY name");
        cmd.Parameters.AddWithValue(tenantId);
        var list = new List<WidgetDef>();
        await using var r = await cmd.ExecuteReaderAsync(ct);
        while (await r.ReadAsync(ct)) list.Add(Read(r));
        return list;
    }

    public async Task<WidgetDef?> GetAsync(string tenantId, string id, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand($"SELECT {Cols} FROM docframe_dashboard_widgets WHERE tenant_id = $1 AND id = $2");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(id);
        await using var r = await cmd.ExecuteReaderAsync(ct);
        return await r.ReadAsync(ct) ? Read(r) : null;
    }

    public async Task SaveAsync(WidgetDef w, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand(
            "INSERT INTO docframe_dashboard_widgets (tenant_id, id, name, description, config, created_by, created_at, updated_at, version) " +
            "VALUES ($1,$2,$3,$4,$5::jsonb,$6,$7,$8,$9) ON CONFLICT (tenant_id, id) DO UPDATE SET name=$3, description=$4, config=$5::jsonb, updated_at=$8, version=$9");
        cmd.Parameters.AddWithValue(w.TenantId); cmd.Parameters.AddWithValue(w.Id); cmd.Parameters.AddWithValue(w.Name);
        cmd.Parameters.AddWithValue((object?)w.Description ?? DBNull.Value);
        cmd.Parameters.AddWithValue(JsonSerializer.Serialize(w.Config, DashboardJson.Options));
        cmd.Parameters.AddWithValue(w.CreatedBy); cmd.Parameters.AddWithValue(w.CreatedAt); cmd.Parameters.AddWithValue(w.UpdatedAt); cmd.Parameters.AddWithValue(w.Version);
        await cmd.ExecuteNonQueryAsync(ct);
    }

    public async Task DeleteAsync(string tenantId, string id, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand("DELETE FROM docframe_dashboard_widgets WHERE tenant_id = $1 AND id = $2");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(id);
        await cmd.ExecuteNonQueryAsync(ct);
    }

    public async Task<DashboardLayout> GetLayoutAsync(string tenantId, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand("SELECT layout::text FROM docframe_dashboards WHERE tenant_id = $1");
        cmd.Parameters.AddWithValue(tenantId);
        return await cmd.ExecuteScalarAsync(ct) is string j ? JsonSerializer.Deserialize<DashboardLayout>(j, DashboardJson.Options) ?? new() : new();
    }

    public async Task SaveLayoutAsync(string tenantId, DashboardLayout layout, CancellationToken ct = default)
    {
        await using var cmd = db.CreateCommand(
            "INSERT INTO docframe_dashboards (tenant_id, layout, updated_at) VALUES ($1, $2::jsonb, now()) ON CONFLICT (tenant_id) DO UPDATE SET layout = $2::jsonb, updated_at = now()");
        cmd.Parameters.AddWithValue(tenantId); cmd.Parameters.AddWithValue(JsonSerializer.Serialize(layout, DashboardJson.Options));
        await cmd.ExecuteNonQueryAsync(ct);
    }
}

public static class ReportingServiceCollectionExtensions

    /// <summary>

    {

        configure?.Invoke(options);
        services.AddSingleton(options);
        services.AddSingleton(_ => new ReadOnlyDataSource(NpgsqlDataSource.Create(readOnlyConnectionString)));
        services.AddSingleton(sp => new AnalyticsDataSource(string.IsNullOrWhiteSpace(options.AnalyticsConnectionString)
            ? sp.GetRequiredService<NpgsqlDataSource>()
            : NpgsqlDataSource.Create(options.AnalyticsConnectionString), owns: true));
        services.AddSingleton<ReportingSchemaSync>();
        services.Replace(ServiceDescriptor.Singleton<IReportingQueryExecutor, PostgresReportingQueryExecutor>());
        services.Replace(ServiceDescriptor.Singleton<IWidgetStore, PostgresWidgetStore>());
        services.Replace(ServiceDescriptor.Singleton<Framework.Reports.IReportTemplateStore, PostgresReportTemplateStore>());
        return services;
    }

    /// <summary>
    /// Memeriksa koneksi read-only dashboard saat startup dan menulis PERINGATAN yang jelas bila bermasalah
    /// (mis. role belum dibuat). Tidak menghentikan aplikasi: fitur lain tetap berjalan.
    /// </summary>
    public static async Task<string?> CheckReportingConnectionAsync(this IServiceProvider services, CancellationToken ct = default)
    {
        if (services.GetService<IReportingQueryExecutor>() is not PostgresReportingQueryExecutor exec) return null;
        try
        {
            await exec.ExecuteAsync("__startup_check__", "SELECT 1", [], 1, ct);
            return null;
        }
        catch (Exception ex) when (ex is ReportingUnavailableException or QueryRejectedException)
        {
            services.GetRequiredService<ILoggerFactory>().CreateLogger("Framework.Dashboards")
                .LogWarning("Dashboard SQL tidak bisa dipakai: {Problem}", ex.Message);
            return ex.Message;
        }
    }

    public static Task SyncReportingSchemaAsync(this IServiceProvider services, CancellationToken ct = default) =>
        services.GetService<ReportingSchemaSync>()?.SyncAsync(ct) ?? Task.CompletedTask;
}

using System.Collections.Concurrent;
using System.Security.Cryptography;
using Framework.Reports;

namespace Framework.Web;

/// <summary>
/// File laporan yang sudah dibuat di sirkuit Blazor diambil browser lewat /report-downloads/{token}: token acak,
/// sekali pakai, terikat pengguna & tenant, kedaluwarsa 10 menit. File besar tidak dikirim lewat SignalR dan
/// parameter laporan tidak muncul di URL.
/// </summary>
public sealed class ReportDownloads(TimeProvider clock)
{
    private sealed record Item(string TenantId, string UserId, ReportOutput Output, DateTimeOffset Expires);
    private readonly ConcurrentDictionary<string, Item> _items = new();

    public string Put(string tenantId, string userId, ReportOutput output)
    {
        var now = clock.GetUtcNow();
        foreach (var kv in _items.Where(kv => kv.Value.Expires < now).ToList()) _items.TryRemove(kv.Key, out _);
        var token = Convert.ToHexString(RandomNumberGenerator.GetBytes(24)).ToLowerInvariant();
        _items[token] = new Item(tenantId, userId, output, now.AddMinutes(10));
        return token;
    }

    public ReportOutput? Take(string token, string tenantId, string userId)
    {
        if (!_items.TryRemove(token, out var item)) return null;
        return item.Expires >= clock.GetUtcNow() && item.TenantId == tenantId && item.UserId == userId ? item.Output : null;
    }
}

    private static readonly (int, string)[] Steps = [(1, "Query"), (2, "Parameter & pratinjau"), (3, "Template"), (4, "Simpan")];
    private ReportTemplateDef def = new();
    private int step = 1, maxStep = 1;

    private TemplateCheck? check;

    private static readonly IEnumerable<SelectOption<string>> TypeOptions =
        [new(ParamTypes.Text, "Teks"), new(ParamTypes.Number, "Angka"), new(ParamTypes.Date, "Tanggal"), new(ParamTypes.Boolean, "Boolean")];

    private static IEnumerable<SelectOption<string>> BindingOptions(WidgetParameter p)
    {
        yield return new(ParamBindings.Default, "Diisi saat membuat laporan");
        if (p.Type == ParamTypes.Text) { yield return new(ParamBindings.UserDepartment, "Departemen pengguna"); yield return new(ParamBindings.UserId, "ID pengguna"); }
    }

    protected override async Task OnParametersSetAsync()
    {
        if (TemplateId == loadedKey && loadedKey is not null) return;
        loadedKey = TemplateId ?? "";
        if (TemplateId is null || !ReportTemplateService.CanDesign(Current.Required)) return;
        if (await Reports.GetAsync(Current.Required, TemplateId) is not { } d) { (errorTitle, error) = ("Tidak ditemukan", "Template laporan tidak ditemukan."); return; }
        def = d; isEdit = true; maxStep = 2;
        template = await Reports.GetTemplateFileAsync(Current.Required, d.Id);
        templateName = d.TemplateFileName;
    }

    private void GoTo(int n) { if (n <= maxStep) { step = n; error = null; } }

    private void Insert(string text)
    {
        var sql = def.Config.Sql.TrimEnd();
        def.Config.Sql = sql + (sql.Length == 0 ? "" : " ") + text;
    }

    private void Detect()
    {
        error = null;
        var (err, ps) = ReportTemplateService.DetectParameters(def.Config.Sql, def.Config.HeaderSql, def.Config.Parameters);
        if (err is not null) { (errorTitle, error) = ("Query belum bisa dipakai", err); return; }
        def.Config.Parameters = ps;
        preview = null; check = null;
        step = maxStep = 2;
    }

    private async Task RunPreview()
    {
        error = null; busy = true;
        try
        {
            preview = await Reports.PreviewAsync(Current.Required, def.Config);
            maxStep = Math.Max(maxStep, 3);
            Recheck();
        }
        catch (Exception ex) when (ex is QueryRejectedException or ReportingUnavailableException)
        {
            preview = null; maxStep = 2;
            (errorTitle, error) = ("Query gagal dijalankan", ex.Message);
        }
        finally { busy = false; }
    }

    private string StarterUrl => preview is null ? "#"
        : $"data:{ReportFormats.ContentType(def.Config.Format)};base64,{Convert.ToBase64String(ReportTemplateService.Starter(def.Config, string.IsNullOrWhiteSpace(def.Name) ? "Laporan" : def.Name, preview))}";

    private void SetFormat(string f)
    {
        if (def.Config.Format == f) return;
        def.Config.Format = f;
        if (template is not null) { template = null; templateName = null; templateIsNew = false; check = null; tryUrl = null; }
    }

    private async Task Upload(InputFileChangeEventArgs e)
    {
        error = null; tryUrl = null;
        if (e.File.Size > OoxmlPackage.MaxTemplateBytes) { (errorTitle, error) = ("File terlalu besar", "Template maksimal 5 MB."); return; }
        await using var s = e.File.OpenReadStream(OoxmlPackage.MaxTemplateBytes);
        using var ms = new MemoryStream();
        await s.CopyToAsync(ms);
        template = ms.ToArray(); templateIsNew = true; templateName = e.File.Name;
        Recheck();
    }

    private void Recheck()
    {
        check = template is null || preview is null ? null : ReportTemplateService.CheckTemplate(def.Config, template, preview);
        if (check is { Problems.Count: 0 }) maxStep = Math.Max(maxStep, 4);
    }

    private async Task TryRender()
    {
        error = null; busy = true;
        try
        {
            var output = await Reports.TryRenderAsync(Current.Required, string.IsNullOrWhiteSpace(def.Name) ? "uji" : def.Name, def.Config, template!);
            (tryUrl, tryName) = ($"data:{output.ContentType};base64,{Convert.ToBase64String(output.Content)}", output.FileName);
        }
        catch (Exception ex) when (ex is TemplateException or QueryRejectedException or ReportingUnavailableException)
        {
            (errorTitle, error) = ("Uji coba gagal", ex.Message);
        }
        finally { busy = false; }
    }

    private void ToggleRole(string role, bool on)
    {
        if (on && !def.Config.Roles.Contains(role)) def.Config.Roles.Add(role);
        if (!on) def.Config.Roles.Remove(role);
    }

    private async Task Save()
    {
        error = null; busy = true;
        try
        {
            if (templateIsNew) def.TemplateFileName = templateName ?? "template." + def.Config.Format;
            var saved = await Reports.SaveAsync(Current.Required, def, templateIsNew ? template : null);

        }
        catch (Exception ex) when (ex is TemplateException or QueryRejectedException or ReportingUnavailableException or PermissionDeniedException)
        {
            (errorTitle, error) = ("Template belum bisa disimpan", ex.Message);
        }
        finally { busy = false; }
    }

    private static string? LibreOffice(byte[] file, string ext, string to)
    {
        if (!File.Exists("/usr/bin/soffice")) return null;
        var dir = Directory.CreateTempSubdirectory("lo").FullName;
        var src = Path.Combine(dir, "doc." + ext);
        File.WriteAllBytes(src, file);
        var psi = new ProcessStartInfo("soffice", ["--headless", "--norestore", "--convert-to", to, "--outdir", dir, src]) { RedirectStandardOutput = true, RedirectStandardError = true };
        psi.Environment["HOME"] = "/tmp/lohome";
        using var p = Process.Start(psi)!;
        p.WaitForExit(90_000);
        var outFile = Directory.GetFiles(dir).FirstOrDefault(f => f != src);
        return outFile is null ? throw new Exception("LibreOffice gagal membuka file: " + p.StandardError.ReadToEnd()) : File.ReadAllText(outFile);
    }

        byte[] With(byte[] src, string name, byte[] content)
        {
            var ms = new MemoryStream(); ms.Write(src);
            using (var z = new ZipArchive(ms, ZipArchiveMode.Update, true)) { using var s = z.CreateEntry(name).Open(); s.Write(content); }
            return ms.ToArray();
        }
        Assert.Throws<TemplateException>(() => OoxmlPackage.Open(With(good, "xl/vbaProject.bin", [1, 2, 3]), ReportFormats.Xlsx), "template bermakro ditolak");
        Assert.Throws<TemplateException>(() => OoxmlPackage.Open(Encoding.UTF8.GetBytes("bukan zip"), ReportFormats.Xlsx), "bukan ZIP ditolak");
        Assert.Throws<TemplateException>(() => OoxmlPackage.Open(good, ReportFormats.Docx), "xlsx diunggah sebagai Word ditolak");
        var bomb = With(good, "xl/media/bomb.bin", new byte[OoxmlPackage.MaxUncompressedBytes + 1]);
        Assert.True(bomb.Length < OoxmlPackage.MaxTemplateBytes, "uji zip bomb: file kecil, isi besar");
        Assert.Throws<TemplateException>(() => OoxmlPackage.Open(bomb, ReportFormats.Xlsx), "zip bomb ditolak");
        var xxe = OoxmlPackage.FromParts([("[Content_Types].xml", "<Types/>"), ("xl/workbook.xml", "<?xml version=\"1.0\"?><!DOCTYPE x [<!ENTITY e SYSTEM \"file:///etc/passwd\">]><workbook>&e;</workbook>")]).Save();
        Assert.Throws<TemplateException>(() => XlsxTemplate.Scan(OoxmlPackage.Open(xxe, ReportFormats.Xlsx)), "DTD / XXE ditolak");
        var problems = TemplateTags.Validate(TemplateTags.Find("{{Tidakada}} [[kolom_x]] {{Sum:nama}} {{Rata:gaji}} {{Count}} [[No]] {{Today}}", "uji"),
            Gaji().Columns, null, ["Periode"]);
        Assert.Equal(4, problems.Count, "tag tidak dikenal, kolom tidak ada, Sum kolom teks, fungsi tidak dikenal: " + string.Join(" | ", problems));
        return Task.CompletedTask;
    }
}

    [Test, ReportingTest]
    public async Task Laporan_template_disimpan_lalu_dibuat_dengan_parameter_dan_role()
    {
        var admin = U("admin");
        foreach (var format in new[] { ReportFormats.Xlsx, ReportFormats.Docx })
        {
            var config = PengajuanConfig(format);
            var (err, ps) = ReportTemplateService.DetectParameters(config.Sql, config.HeaderSql, config.Parameters);
            Assert.True(err is null && ps.Single().Name == "Dept", "parameter dari query detail & header");
            var preview = await Reports.PreviewAsync(admin, config);
            Assert.True(preview.Detail.Rows.Count == 3 && preview.Header!.Rows.Count == 1, "pratinjau detail (IT) & header");
            var starter = ReportTemplateService.Starter(config, "Daftar Pengajuan {{Dept}}", preview);
            Assert.Equal(0, ReportTemplateService.CheckTemplate(config, starter, preview).Problems.Count, "template contoh langsung cocok");

            var saved = await Reports.SaveAsync(admin, new ReportTemplateDef { Name = $"Pengajuan ({format})", Config = config, TemplateFileName = "contoh." + format }, starter);
            Assert.True(saved.Id.StartsWith("rt_") && saved.Version == 1, "tersimpan");
            await Assert.Throws<PermissionDeniedException>(() => Reports.GenerateAsync(U("andi"), saved.Id, new Dictionary<string, string?>()), "role lain ditolak");
            Assert.True((await Reports.ListAsync(U("andi"))).All(d => d.Id != saved.Id), "tidak tampil di daftar role lain");

            var output = await Reports.GenerateAsync(U("dewi"), saved.Id, new Dictionary<string, string?> { ["Dept"] = "Marketing" });
            Assert.True(output.Rows == 2 && output.FileName.StartsWith("Pengajuan Marketing 20") && output.FileName.EndsWith("." + format), $"parameter dipakai: {output.FileName}, {output.Rows} baris");
            var text = format == ReportFormats.Xlsx ? LibreOffice(output.Content, "xlsx", "csv") : LibreOffice(output.Content, "docx", "txt:Text");
            if (text is not null)
                Assert.True(text.Contains("PR-2026-00004") && text.Contains("Sari Wulandari") && !text.Contains("PR-2026-00001") && text.Contains("PT Acme Indonesia"),
                    $"isi {format} sesuai parameter: " + text);
        }
    }

    [Test, ReportingTest]
    public async Task Laporan_template_tidak_cocok_ditolak_dan_hasil_terlalu_besar_tidak_dipotong()
    {
        var admin = U("admin");
        var config = PengajuanConfig(ReportFormats.Xlsx);
        var wrong = XlsxTemplate.Starter("x {{TidakAda}}", [new("kolom_lama", "text")], null);
        var ex = await Assert.Throws<TemplateException>(() => Reports.SaveAsync(admin, new ReportTemplateDef { Name = "Salah", Config = config }, wrong), "template tidak cocok");
        Assert.True(ex.Message.Contains("kolom_lama") && ex.Message.Contains("TidakAda"), "pesan menyebut tag yang bermasalah: " + ex.Message);
        await Assert.Throws<PermissionDeniedException>(() => Reports.SaveAsync(U("dewi"), new ReportTemplateDef { Name = "X", Config = config }, wrong), "hanya System Manager merancang");

        var preview = await Reports.PreviewAsync(admin, config);
        var saved = await Reports.SaveAsync(admin, new ReportTemplateDef { Name = "Kecil", Config = config }, ReportTemplateService.Starter(config, "x", preview));
        var tiny = new ReportTemplateService(sp.GetRequiredService<IReportTemplateStore>(), sp.GetRequiredService<IReportingQueryExecutor>(), TimeProvider.System,
            new ReportLimits { MaxRowsExcel = 2 });
        var big = await Assert.Throws<TemplateException>(() => tiny.GenerateAsync(admin, saved.Id, new Dictionary<string, string?> { ["Dept"] = "IT" }), "3 baris > batas 2");
        Assert.True(big.Message.Contains("melebihi"), "pesan meminta mempersempit parameter: " + big.Message);
        var ok = await tiny.GenerateAsync(admin, saved.Id, new Dictionary<string, string?> { ["Dept"] = "Marketing" });
        Assert.Equal(2, ok.Rows, "tepat di batas tetap dibuat");
    }

- Butuh mode PostgreSQL + koneksi read-only (sama seperti Dashboard). Tanpa itu
  halaman menampilkan pemberitahuan.
- File template disimpan di database (kolom bytea, maks. 5 MB).
- Laporan berisi data seluruh tenant: aturan "milik sendiri" tidak berlaku di SQL.
  Batasi role, atau ikat parameter ke "Departemen pengguna".
- Batasan: tag di dalam rumus Excel tidak diproses; rumus yang merujuk sheet LAIN
  yang ikut melebar tidak disesuaikan; satu query detail per template.
- Konfigurasi: Reports:MaxRowsExcel (50000), Reports:MaxRowsWord (5000),
  Reports:StatementTimeoutSeconds (30).
